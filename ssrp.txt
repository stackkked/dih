local function safeCall(fn, ...)
    if type(fn) ~= "function" then
        return nil
    end

local function teleportCharacterToCFrame(targetCFrame)
    local character = LocalPlayer.Character
    if not character then
        return false
    end
    if character:IsA("Model") then
        character:PivotTo(targetCFrame)
        return true
    end
    local root = character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart
    if root then
        root.CFrame = targetCFrame
        return true
    end
    return false
end

local function getKnitRemote(serviceName, remoteName)
    local packages = ReplicatedStorage:FindFirstChild("Packages")
    if not packages then
        return nil
    end
    local index = packages:FindFirstChild("_Index")
    if not index then
        return nil
    end
    for _, child in ipairs(index:GetChildren()) do
        local knitFolder = child:FindFirstChild("knit")
        local services = knitFolder and knitFolder:FindFirstChild("Services")
        local service = services and services:FindFirstChild(serviceName)
        local rfFolder = service and service:FindFirstChild("RF")
        local remote = rfFolder and rfFolder:FindFirstChild(remoteName)
        if remote then
            return remote
        end
    end
    return nil
end

local function resolveSampleInteractTarget()

    local parts = workspace:GetPartBoundsInBox(AUTO_PRESTIGE_SAMPLE_BOX_CFRAME, Vector3.new(3, 3, 3), overlapParams)

local function getTeamPrestigeGoal()
    local team = LocalPlayer.Team
    local osirisTeam = Teams:FindFirstChild("Osiris Collective")
    local researchSubjectTeam = Teams:FindFirstChild("Research Subject")

    if team and osirisTeam and team == osirisTeam then
        return 1920
    end

    if team and researchSubjectTeam and team == researchSubjectTeam then
        return 960
    end

    if team and team.Name == "Osiris Collective" then
        return 1920
    end

    return 960
end

local function readTeamXPValue()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    local mainUi = playerGui and playerGui:FindFirstChild("MainUI")
    local teamXP = mainUi and mainUi:FindFirstChild("TeamXP")
    local xpLabel = teamXP and teamXP:FindFirstChild("XP")
    if not xpLabel then
        return nil
    end

    local rawText = tostring(xpLabel.Text or xpLabel:GetAttribute("Text") or "")
    local currentText = rawText:match("^(%d+)%s*/")
    if currentText then
        return tonumber(currentText) or nil, rawText
    end

    currentText = rawText:match("^(%d+)")
    if currentText then
        return tonumber(currentText) or nil, rawText
    end

    return nil, rawText
end

local function startAutoPrestigeLoop()
    if osirisState.AutoPrestigeRunning then
        return
    end

    osirisState.AutoPrestigeRunning = true
    osirisState.AutoPrestigeNeedsTeleport = true
    task.spawn(function()
        task.spawn(function()
            while osirisState.AutoPrestige do
                local interactRF, sellRF = getAutoPrestigeRemotes()

                if not interactRF or not sellRF then
                    task.wait(0.5)
                elseif osirisState.AutoPrestigeNeedsTeleport then
                    teleportCharacterToCFrame(AUTO_PRESTIGE_TELEPORT_CFRAME)
                    osirisState.AutoPrestigeNeedsTeleport = false
                    task.wait(0.1)
                elseif osirisState.AutoPrestigeActionIsInteract then
                    local sampleTarget = resolveSampleInteractTarget()
                    if sampleTarget then
                        pcall(function()
                            interactRF:InvokeServer(sampleTarget)
                        end)
                        osirisState.AutoPrestigeActionIsInteract = false
                    else
                        task.wait(0.05)
                    end
                else
                    pcall(function()
                        sellRF:InvokeServer()
                    end)
                    osirisState.AutoPrestigeActionIsInteract = true
                end
            end
        end)

        while osirisState.AutoPrestige do
            local _, _, prestigeRF = getAutoPrestigeRemotes()
            if not prestigeRF then
                task.wait(0.5)
            else
                local teamGoal = getTeamPrestigeGoal()
                local currentXP = readTeamXPValue()
                if currentXP and currentXP >= teamGoal then
                    pcall(function()
                        prestigeRF:InvokeServer()
                    end)
                    osirisState.AutoPrestigeActionIsInteract = true
                    osirisState.AutoPrestigeNeedsTeleport = true
                    task.wait(2.5)
                end

                task.wait(0.2)
            end
        end

        osirisState.AutoPrestigeRunning = false
    end)
end

do

    if mt and setreadonly and newcclosure and getnamecallmethod then
        local oldNamecall = mt.__namecall
        setreadonly(mt, false)

        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            local args = { ... }

            if method == "FireServer" then
                if self.Name == "SentryRemote" then
                    return
                end

                if self.Name == "ReplicateHit" then
                    local now = os.clock()
                    for i = #hitHistory, 1, -1 do
                        if now - hitHistory[i] > 3 then
                            table.remove(hitHistory, i)
                        end
                    end
                    if #hitHistory >= 30 then
                        return
                    end
                    table.insert(hitHistory, now)
                end
            end

            if method == "GetAttribute" then
                local attr = args[1]

                if attr == "Oxygen" then return 5000 end
                if attr == "Integrity" then return 100 end
                if attr == "RunBlocked" then return false end
                if attr == "IsRagdoll" then return false end
                if attr == "Cuffed" then return false end
                if attr == "Tased" then return false end
                if attr == "Trespassing" then return false end
            end

            return oldNamecall(self, ...)
        end)

        setreadonly(mt, true)
    end
end

local function toggleFloatingWindow(win)
    if not win then
        return
    end
    if win.Show and win.Hide and win._Visible ~= nil then
        pcall(function()
            if win._Visible then
                win:Hide()
            else
                win:Show()
            end
        end)
        return
    end
    if win.Toggle then
        pcall(function()
            win:Toggle()
        end)
        return
    end
    if win._Frame then
        win._Frame.Visible = not win._Frame.Visible
    end
end

local function restoreInstantInteract()
    local controller = getInteractionController()
    if controller then
        if interactionState.OriginalGetConfig then
            controller._getInteractConfig = interactionState.OriginalGetConfig
        end
        if interactionState.OriginalShowHUD then
            controller.ShowHUD = interactionState.OriginalShowHUD
        end
        if type(controller._configCache) == "table" then
            table.clear(controller._configCache)
        end
    end
    interactionState.Applied = false
end

local function applyInstantInteract()
    if not interactionState.Enabled then return false end
    local controller = getInteractionController()
    if not controller then return false end

    if interactionState.OriginalGetConfig == nil then
        interactionState.OriginalGetConfig = controller._getInteractConfig
    end
    if interactionState.OriginalShowHUD == nil then
        interactionState.OriginalShowHUD = controller.ShowHUD
    end

    if type(controller._configCache) == "table" then
        table.clear(controller._configCache)
    end

    if type(interactionState.OriginalGetConfig) == "function" then
        controller._getInteractConfig = function(self, interactablePart)
            local config = interactionState.OriginalGetConfig(self, interactablePart)
            if config and type(config) == "table" then
                config.Duration = 0
            end
            return config
        end
    end

    if type(interactionState.OriginalShowHUD) == "function" then
        controller.ShowHUD = function(self, key, part, callback, topText, bottomText, duration, disabled, keysOverride, isInteractingCallback)
            return interactionState.OriginalShowHUD(self, key, part, callback, topText, bottomText, 0, disabled, keysOverride, isInteractingCallback)
        end
    end

    interactionState.Applied = true
    return true
end

local function setInstantInteract(enabled)
    interactionState.Enabled = enabled and true or false
    if interactionState.Enabled then
        applyInstantInteract()
        if not interactionState.WatcherRunning then
            interactionState.WatcherRunning = true
            task.spawn(function()
                while interactionState.Enabled do
                    applyInstantInteract()
                    task.wait(0.25)
                end
                restoreInstantInteract()
                interactionState.WatcherRunning = false
            end)
        end
    else
        restoreInstantInteract()
    end
end

local function cleanCodeString(str)
    if not str then return nil end
    return tostring(str):gsub("%s+", ""):gsub("%-", ""):upper()
end

local function parseTargetConnections(page)
    local connections = {}
    local function isTerminalString(val) return type(val) == "string" and val:sub(1, 4) == "Wire" end
    local function search(tbl)
        for k, v in pairs(tbl) do
            if isTerminalString(k) and isTerminalString(v) then
                connections[k] = v
            elseif type(k) == "string" and type(v) == "string" and k:sub(1, 4) == "Wire" and v:sub(1, 4) == "Wire" then
                local top = k:match("Top$") and k or (k .. "Top")
                local bot = v:match("Bottom$") and v or (v .. "Bottom")
                connections[top] = bot
            elseif (type(k) == "number" or tonumber(k)) and (type(v) == "number" or tonumber(v)) then
                local numKey = tonumber(k)
                local numVal = tonumber(v)
                if numKey >= 1 and numKey <= 4 and numVal >= 1 and numVal <= 4 then
                    connections["Wire" .. numKey .. "Top"] = "Wire" .. numVal .. "Bottom"
                end
            elseif type(v) == "table" and k ~= "Parent" then
                search(v)
            end
        end
    end
    search(page)
    return connections
end

local POWERBOX_TURRET_GOOD_COLOR = Color3.fromRGB(39, 70, 45)
local POWERBOX_TURRET_BAD_COLOR = Color3.fromRGB(52, 142, 64)
local POWERBOX_TURRET_SWITCH_NAMES = {"Switch1", "Switch2", "Switch3", "Switch4"}
local POWERBOX_TURRET_ACTION_DELAY = 0.5

local function readAppearanceColorValue(switchPart)
    if not switchPart then
        return nil
    end
    return switchPart.Color
end

local function colorClose(a, b)
    if not (a and b) then
        return false
    end
    return math.abs(a.R - b.R) <= 0.01
        and math.abs(a.G - b.G) <= 0.01
        and math.abs(a.B - b.B) <= 0.01
end

local function getNearestPowerboxTurret(originPosition)
    local bestModel = nil
    local bestDistance = math.huge
    for _, inst in ipairs(workspace:GetDescendants()) do
        if inst:IsA("Model") and inst.Name == "PowerboxTurret" then
            local root = inst.PrimaryPart
                or inst:FindFirstChild("Switch1")
                or inst:FindFirstChildWhichIsA("BasePart")
            if root and originPosition then
                local distance = (root.Position - originPosition).Magnitude
                if distance < bestDistance then
                    bestDistance = distance
                    bestModel = inst
                end
            elseif not bestModel then
                bestModel = inst
            end
        end
    end
    return bestModel
end

local function getTurretSwitchSnapshot(turretModel)
    local snapshot = {}
    if not turretModel then
        return snapshot
    end
    for _, switchName in ipairs(POWERBOX_TURRET_SWITCH_NAMES) do
        local switchPart = turretModel:FindFirstChild(switchName, true)
        if switchPart and switchPart:IsA("BasePart") then
            snapshot[switchName] = readAppearanceColorValue(switchPart) or switchPart.Color
        end
    end
    return snapshot
end

local function sawGoodColorTransition(beforeSnapshot, afterSnapshot)
    for switchName, afterColor in pairs(afterSnapshot) do
        if colorClose(afterColor, POWERBOX_TURRET_GOOD_COLOR) then
            if not colorClose(beforeSnapshot[switchName], POWERBOX_TURRET_GOOD_COLOR) then
                return true, switchName
            end
        end
    end
    return false
end

local function getCanonicalTurretConnectionKey(firstTerminal, secondTerminal)
    firstTerminal = tostring(firstTerminal)
    secondTerminal = tostring(secondTerminal)
    if firstTerminal > secondTerminal then
        firstTerminal, secondTerminal = secondTerminal, firstTerminal
    end
    return firstTerminal .. ">" .. secondTerminal
end

local function markTurretSolvedSwitches(snapshot, solvedSwitches)
    for switchName, color in pairs(snapshot) do
        if colorClose(color, POWERBOX_TURRET_GOOD_COLOR) then
            solvedSwitches[switchName] = true
        end
    end
end

local function resetTurretWires(substationService, powerboxId, wireData)
    if type(wireData) ~= "table" then
        return false
    end

    local touched = {}
    local resetAny = false

    for wireName, connectedName in pairs(wireData) do
        if type(wireName) == "string" and wireName:sub(1, 4) == "Wire" and type(connectedName) == "string" then
            if not touched[wireName] and not touched[connectedName] then
                safeCall(substationService.WireInteraction, substationService, powerboxId, wireName)
                touched[wireName] = true
                touched[connectedName] = true
                resetAny = true
                task.wait(POWERBOX_TURRET_ACTION_DELAY)
            end
        end
    end

    if resetAny then
        task.wait(POWERBOX_TURRET_ACTION_DELAY)
    end

    return resetAny
end

local function getExecutiveOrderTarget(payload)
    if not payload then return nil end
    if payload.TargetPlayer and payload.TargetPlayer:IsA("Player") then return payload.TargetPlayer end
    if type(payload.TargetName) == "string" then
        local p = Players:FindFirstChild(payload.TargetName)
        if p then return p end
    end
    if type(payload.Target) == "string" then
        local p = Players:FindFirstChild(payload.Target)
        if p then return p end
    end
    if payload.Target and typeof(payload.Target) == "Instance" and payload.Target:IsA("Player") then
        return payload.Target
    end
    local desc = tostring(payload.Description or ""):upper()
    local title = tostring(payload.Name or ""):upper()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local name = player.Name:upper()
            local displayName = player.DisplayName:upper()
            if (#name >= 3 and (desc:find(name, 1, true) or title:find(name, 1, true))) or
               (#displayName >= 3 and (desc:find(displayName, 1, true) or title:find(displayName, 1, true))) then
                return player
            end
        end
    end
    return nil
end

local function isPartVisible(part, character)
    local camera = workspace.CurrentCamera
    if not camera then return false end
    local origin = camera.CFrame.Position
    local destination = part.Position
    local direction = destination - origin
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = { LocalPlayer.Character, camera, character }
    raycastParams.IgnoreWater = true
    
    local result = workspace:Raycast(origin, direction, raycastParams)
    return result == nil
end

local function getAnyVisiblePart(character)
    local camera = workspace.CurrentCamera
    if not camera then return nil end
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("BasePart") and child.Name ~= "HumanoidRootPart" then
            local _, onScreen = camera:WorldToViewportPoint(child.Position)
            if onScreen and isPartVisible(child, character) then return child end
        end
    end
    return nil
end

        end
    end

    if FirearmController.Stats then FirearmController.Stats.RangeLimit = 99999 end
    FirearmController.RangeHardLimit = 99999

    if FirearmController.BulletData then
        for _, bullet in pairs(FirearmController.BulletData) do
            if type(bullet) == "table" and bullet.Velocity then
                bullet.Velocity = 999999
            end
        end
    end

    local function getClosestTargetInFOV()
        local myChar = LocalPlayer.Character
        if not myChar or not myChar.PrimaryPart then return nil end

        local closestPart = nil
        local shortestDist = weaponState.FOVRadius
        local mousePos = UserInputService:GetMouseLocation()
        local camera = workspace.CurrentCamera
        if not camera then return nil end
        
        local maxRange = (FirearmController and FirearmController.Stats and FirearmController.Stats.RangeLimit) or 99999

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
                local canDamage = true
                if type(FirearmController.CanDamage) == "function" then
                    canDamage = FirearmController:CanDamage(player)
                end
                
                if canDamage then
                    if player.Team and player.Team.Name == "Health & Research" then
                        continue
                    end

                    local char = player.Character
                    local root = char.PrimaryPart or char:FindFirstChild("Head")
                    if root then
                        local distance3D = (root.Position - myChar.PrimaryPart.Position).Magnitude
                        if distance3D <= maxRange then
                            local rootPos, onScreen = camera:WorldToViewportPoint(root.Position)
                            if onScreen then
                                local rootDist = (Vector2.new(rootPos.X, rootPos.Y) - mousePos).Magnitude
                                if rootDist < shortestDist + 150 then
                                    local chosenPart = nil
                                    local chosenDist = math.huge
                                    
                                    for partName, enabled in pairs(weaponState.AimParts) do
                                        if enabled then
                                            local part = char:FindFirstChild(partName)
                                            if part then
                                                local pos, partOnScreen = camera:WorldToViewportPoint(part.Position)
                                                if partOnScreen and isPartVisible(part, char) then
                                                    local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                                                    if dist < chosenDist then
                                                        chosenDist = dist
                                                        chosenPart = part
                                                    end
                                                end
                                            end
                                        end
                                    end
                                    
                                    if not chosenPart then
                                        local fallbackPart = getAnyVisiblePart(char)
                                        if fallbackPart then
                                            local pos, fbOnScreen = camera:WorldToViewportPoint(fallbackPart.Position)
                                            if fbOnScreen then
                                                chosenPart = fallbackPart
                                                chosenDist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                                            end
                                        end
                                    end
                                    
                                    if chosenPart and chosenDist < shortestDist then
                                        shortestDist = chosenDist
                                        closestPart = chosenPart
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
        return closestPart
    end

    local oldFire = FirearmController.Fire
    if type(oldFire) == "function" then
        FirearmController.Fire = function(self, arg1, arg2)
            local origSpread = nil
            local origCCF = nil
            local origShakeOnce = nil
            
            if weaponState.NoSpread and self.Stats and self.Stats.Bullet then
                origSpread = self.Stats.Bullet.BuckshotSpread
                self.Stats.Bullet.BuckshotSpread = 0
            end
            
            if weaponState.NoRecoil and self.CameraController and self.CameraController.CameraShaker then
                origShakeOnce = self.CameraController.CameraShaker.ShakeOnce
                self.CameraController.CameraShaker.ShakeOnce = function() end
            end
            
            if weaponState.SilentAim then
                local instantTarget = getClosestTargetInFOV()
                if instantTarget then
                    origCCF = self.CameraController.CCF
                    self.CameraController.CCF = CFrame.lookAt(origCCF.Position, instantTarget.Position)
                end
            end
            
            local result = oldFire(self, arg1, arg2)
            
            if origCCF then self.CameraController.CCF = origCCF end
            if origSpread then self.Stats.Bullet.BuckshotSpread = origSpread end
            if origShakeOnce then self.CameraController.CameraShaker.ShakeOnce = origShakeOnce end
            
            return result
        end
    end
end)

        if espState.PowerBoxESP then updatePowerboxESP() end
    end

task.spawn(function()
    while task.wait(0.5) do
        if not espState.Chams then continue end
        
        local FirearmController = nil
        if Knit and type(Knit.GetController) == "function" then
            FirearmController = safeCall(Knit.GetController, "FirearmController")
        end
        
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer then continue end
            
            local char = player.Character
            if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0 then
                local isEnemy = true
                if FirearmController and type(FirearmController.CanDamage) == "function" then
                    isEnemy = FirearmController:CanDamage(player)
                end
                
                if isEnemy then
                    local teamColor = player.Team and player.Team.TeamColor.Color or Color3.fromRGB(255, 255, 255)
                    local isEOTarget = (player == weaponState.EOTarget)
                    
                    local fillColor = teamColor
                    local outlineColor = Color3.fromRGB(255, 255, 255)
                    local fillTrans = 0.5
                    local outlineTrans = 0.2
                    
                    if isEOTarget then
                        fillColor = Color3.fromRGB(255, 215, 0)
                        outlineColor = Color3.fromRGB(255, 0, 0)
                        fillTrans = 0.2
                        outlineTrans = 0
                    end

                    local hl = espObjects[player]
                    if not hl or hl.Parent ~= char then
                        if hl then hl:Destroy() end
                        hl = Instance.new("Highlight")
                        hl.Name = "FrederikChams"
                        hl.FillColor = fillColor
                        hl.OutlineColor = outlineColor
                        hl.FillTransparency = fillTrans
                        hl.OutlineTransparency = outlineTrans
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.Parent = char
                        espObjects[player] = hl
                    else
                        hl.FillColor = fillColor
                        hl.OutlineColor = outlineColor
                        hl.FillTransparency = fillTrans
                        hl.OutlineTransparency = outlineTrans
                    end
                else
                    local hl = espObjects[player]
                    if hl then hl:Destroy(); espObjects[player] = nil end
                end
            else
                local hl = espObjects[player]
                if hl then hl:Destroy(); espObjects[player] = nil end
            end
        end
    end
end)
