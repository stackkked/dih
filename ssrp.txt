-- 1. ОПТИМИЗИРОВАННЫЙ СПУФИНГ ЧТЕНИЯ (__index)
mt.__index = newcclosure(function(self, key)
    if not checkcaller() then
        -- Спуфинг атрибутов (без рекурсии через rawequal)
        if rawequal(self, Terrain) and FakeStore[key] ~= nil then
            return FakeStore[key]
        end

        -- Маскировка объектов (только если ключ ClassName)
        if key == "ClassName" or key == "className" then
            local class = oldIndex(self, "ClassName")
            if class == "BodyVelocity" or class == "BodyGyro" or class == "AlignPosition" then
                return "Part"
            end
        end
    end
    return oldIndex(self, key)
end)

-- 2. БЛОКИРОВКА ЗАПИСИ ДЕТЕКТОВ (__newindex)
mt.__newindex = newcclosure(function(t, k, v)
    if not checkcaller() then
        if rawequal(t, Terrain) and k == "TerrainCharacter" then
            local val = tostring(v)
            if BAN_CODES[val] or val:find("^9,") then
                warn("!!! ПЕРЕХВАЧЕН КОД БАНА: " .. val)
                FakeStore[k] = -1 -- Врем античиту, что записали -1
                return nil 
            end
        end
    end
    return oldNewIndex(t, k, v)
end)

-- 3. БЫСТРЫЙ NAME_CALL (RemoteEvents + SetAttribute)
mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if not checkcaller() then
        -- Спуфинг GetAttribute
        if method == "GetAttribute" or method == "getAttribute" then
            if rawequal(self, Terrain) and FakeStore[args[1]] ~= nil then
                return FakeStore[args[1]]
            end
        end

        -- Блокировка SetAttribute (Коды бана)
        if method == "SetAttribute" or method == "setAttribute" then
            if rawequal(self, Terrain) and args[1] == "TerrainCharacter" then
                local val = tostring(args[2])
                if BAN_CODES[val] or val:find("^9,") then
                    return nil
                end
            end
        end

        if method == "FireServer" then
            local rName = self.Name
            if BlockedRemotes[rName] or rName:find("Anti") or rName:find("Report") then
                return nil
            end
        end

        if method == "Kick" or method == "kick" then 
            return nil 
        end
    end
    
    return oldNamecall(self, ...)
end)

local function attachFlyScript()
    local character = player.Character
    if not character then
        return false
    end

    local template = getFlyTemplate()
    if not template then
        return false
    end

    purgeFlyRigArtifacts(character)

    local flySpeed = character:FindFirstChild("flyspeed")
    if not flySpeed then
        flySpeed = Instance.new("NumberValue")
        flySpeed.Name = "flyspeed"
        flySpeed.Parent = character
    end
    flySpeed.Value = flyFixedSpeed

    local flyScript = character:FindFirstChild("Fly")
    if not flyScript then
        flyScript = template:Clone()
        flyScript.Name = "Fly"
        flyScript.Disabled = false
        flyScript.Parent = character
    end

    pcall(function()
        flyScript:SetAttribute("Keybind", "E")
    end)
    pcall(function()
        flyScript.Disabled = false
    end)

    return true
end

local function getNearestFlingTarget(maxRange)
    local localCharacter = player.Character
    local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
    if not localRoot then
        return nil, nil, nil
    end

    local bestPlayer, bestRoot, bestDistance
    for _, candidate in ipairs(Players:GetPlayers()) do
        if candidate ~= player then
            local char = candidate.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and root then
                local distance = (root.Position - localRoot.Position).Magnitude
                if distance <= maxRange and (not bestDistance or distance < bestDistance) then
                    bestPlayer = candidate
                    bestRoot = root
                    bestDistance = distance
                end
            end
        end
    end

    return bestPlayer, bestRoot, bestDistance
end

local function updateFlingCircle(hrp, targetFound)
    local cam = workspace.CurrentCamera or camera
    if not cam then
        for _, line in ipairs(flingCircleLines) do
            line.Visible = false
        end
        return
    end

    local color = targetFound and Color3.fromRGB(255, 50, 50) or Color3.fromRGB(0, 255, 255)
    local thickness = targetFound and 3 or 2.25
    local transparency = targetFound and 0.18 or 0.08

    for i = 1, flingCircleSegments do
        local angle1 = (i / flingCircleSegments) * math.pi * 2
        local angle2 = ((i % flingCircleSegments + 1) / flingCircleSegments) * math.pi * 2

        local p1 = hrp.Position + Vector3.new(math.cos(angle1) * _G.FlingRange, -3.2, math.sin(angle1) * _G.FlingRange)
        local p2 = hrp.Position + Vector3.new(math.cos(angle2) * _G.FlingRange, -3.2, math.sin(angle2) * _G.FlingRange)

        local s1, v1 = cam:WorldToViewportPoint(p1)
        local s2, v2 = cam:WorldToViewportPoint(p2)

        local line = flingCircleLines[i]
        if v1 and v2 then
            line.From = Vector2.new(s1.X, s1.Y)
            line.To = Vector2.new(s2.X, s2.Y)
            line.Color = color
            line.Thickness = thickness
            line.Transparency = transparency
            line.Visible = true
        else
            line.Visible = false
        end
    end
end

                    v.CanCollide = false
                end
            end
        end
    end)

    -- Поток сам остановится, когда _G.FlingActive станет false

        local movel = 0.1
        while _G.FlingActive do
            RunService.Heartbeat:Wait()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")

            if character and character.Parent and root and root.Parent then
                local tPlayer, targetRoot = getNearestFlingTarget(_G.FlingRange)
                
                if targetRoot then
                    -- Способ №1: Микро-вибрация CFrame по осям X и Z (Noise в пределах 0.05) для лавинообразных импульсов
                    local jitter = Vector3.new(math.random(-50, 50)/1000, 0, math.random(-50, 50)/1000)
                    
                    pcall(function()
                        root.CFrame = targetRoot.CFrame * CFrame.new(jitter)
                    end)
                    if tPlayer then 
                        targetLog[tPlayer.UserId] = tick() 
                    end
                end

                -- WalkFling из Infinity Yield
                local vel = root.Velocity
                root.Velocity = vel * 10000 + Vector3.new(0, 10000, 0)

                RunService.RenderStepped:Wait()
                if character and character.Parent and root and root.Parent then
                    root.Velocity = vel
                end

                RunService.Stepped:Wait()
                if character and character.Parent and root and root.Parent then
                    root.Velocity = vel + Vector3.new(0, movel, 0)
                    movel = movel * -1
                end
            end
        end
    end)

local function applyFireAxeR15Invisibility(char)
    if not char then
        return false
    end

    local state = fireAxeR15InvisState
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        return false
    end

    if hum.RigType ~= Enum.HumanoidRigType.R15 then
        return false
    end

    if state.Character ~= char then
        state.Character = char
        state.Humanoid = hum
        state.OriginalHipHeight = hum.HipHeight

        if state.Track then
            pcall(function()
                state.Track:Stop()
            end)
            state.Track = nil
        end
    end

    if not state.Animation then
        local animation = Instance.new("Animation")
        animation.AnimationId = "rbxassetid://122954953446602"
        state.Animation = animation
    end

    if not state.Track then
        local ok, track = pcall(function()
            return hum:LoadAnimation(state.Animation)
        end)
        if ok and track then
            state.Track = track
            pcall(function()
                state.Track.Priority = Enum.AnimationPriority.Action4
                state.Track.Looped = true
                state.Track:AdjustSpeed(0.01)
                state.Track:Play()
            end)
        end
    elseif not state.Track.IsPlaying then
        pcall(function()
            state.Track:Play()
            state.Track:AdjustSpeed(0.01)
        end)
    else
        pcall(function()
            state.Track:AdjustSpeed(0.01)
        end)
    end

    pcall(function()
        hum.HipHeight = 0.3
    end)

    return true
end

local function restoreFireAxeR15Invisibility()
    local state = fireAxeR15InvisState

    if state.Track then
        pcall(function()
            state.Track:Stop()
        end)
        state.Track = nil
    end

    if state.Humanoid and state.OriginalHipHeight then
        pcall(function()
            state.Humanoid.HipHeight = state.OriginalHipHeight
        end)
    end

    state.Character = nil
    state.Humanoid = nil
    state.OriginalHipHeight = nil
end

local function stopCustomAnim()
    local state = customAnimState

    if state.Track then
        pcall(function()
            state.Track:Stop(0.15)
        end)

    end

end

local function applyCustomAnim(char)

        return false
    end

    local state = customAnimState
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        return false
    end

    if state.Character ~= char then
        state.Character = char
        state.Humanoid = hum

        if state.Track then
            pcall(function()
                state.Track:Stop(0.1)
            end)
            state.Track = nil
        end
    end

    if not state.Animation then
        local animation = Instance.new("Animation")
        animation.AnimationId = "rbxassetid://5163898852"
        state.Animation = animation
    end

    local animator = hum:FindFirstChildOfClass("Animator")
    if not animator then
        animator = Instance.new("Animator")
        animator.Parent = hum
    end

    if not state.Track then
        local ok, track = pcall(function()
            return animator:LoadAnimation(state.Animation)
        end)
        if ok and track then
            state.Track = track
            pcall(function()
                state.Track.Priority = Enum.AnimationPriority.Action4
                state.Track.Looped = true
                state.Track:AdjustSpeed(1)
                state.Track:Play(0.15, 1, 1)
            end)
        end

            state.Track:Play(0.15, 1, 1)

        end)
    else
        pcall(function()
            state.Track:AdjustSpeed(1)

    return true

local SCP_HELPER_ENABLED = false
local SCP_ESP_STORAGE = {}

local SCP_TARGETS = {
    { name = "SCP-049",  path = {"049"} },
    { name = "SCP-2006", path = {"2006", "2006"} },
    { name = "SCP-017",  path = {"017", "SCP-017"} },
    { name = "SCP-096",  path = {"SCP096"} },
    { name = "SCP-106",  path = {"InteriorBuild", "SCP Items", "SCP106"} },
    { name = "SCP-939-1", path = {"SCPs", "SCP-939-1"} },
    { name = "SCP-939-2", path = {"SCPs", "SCP-939-2"} },
    { name = "SCP-939-3", path = {"SCPs", "SCP-939-3"} },
}

local SCP_COLOR   = Color3.fromRGB(255, 50, 50)
local SCP_OUTLINE = Color3.fromRGB(255, 120, 120)

local function resolveSCPPath(pathTable)
    local current = workspace
    for _, part in ipairs(pathTable) do
        if not current then return nil end
        current = current:FindFirstChild(part)
    end
    return current
end

local function findFirstPart(obj)
    if obj:IsA("BasePart") then return obj end
    if obj:IsA("Model") then
        if obj.PrimaryPart then return obj.PrimaryPart end
        for _, c in ipairs(obj:GetDescendants()) do
            if c:IsA("BasePart") then return c end
        end
    end
    return nil
end

local function cleanSCPESP()
    for obj, data in pairs(SCP_ESP_STORAGE) do
        if data.Highlight then pcall(function() data.Highlight:Destroy() end) end
    end
    SCP_ESP_STORAGE = {}
end

local function createSCPHighlight(obj, scpName)
    local hl = Instance.new("Highlight")
    hl.Name = "SCP_ESP_" .. scpName
    hl.FillColor = SCP_COLOR
    hl.OutlineColor = SCP_OUTLINE
    hl.FillTransparency = 0.75
    hl.OutlineTransparency = 0
    if obj:IsA("Model") then
        hl.Parent = obj
    elseif obj:IsA("BasePart") then
        hl.Adornee = obj
        hl.Parent = CoreGui
    else
        local adornPart = findFirstPart(obj)
        if adornPart then
            hl.Adornee = adornPart
            hl.Parent = CoreGui
        else
            hl:Destroy()
            return nil
        end
    end
    return hl
end

local function rebuildSCPESP()
    cleanSCPESP()
    if not SCP_HELPER_ENABLED then return end
    for _, scpInfo in ipairs(SCP_TARGETS) do
        local obj = resolveSCPPath(scpInfo.path)
        if obj then
            local hl = createSCPHighlight(obj, scpInfo.name)
            if hl then SCP_ESP_STORAGE[scpInfo.name] = { Highlight = hl, Object = obj } end
        end
    end
end

local function updateSCPESP()
    if not SCP_HELPER_ENABLED then return end
    for _, scpInfo in ipairs(SCP_TARGETS) do
        local obj = resolveSCPPath(scpInfo.path)
        if obj then
            if not SCP_ESP_STORAGE[scpInfo.name] then
                local hl = createSCPHighlight(obj, scpInfo.name)
                if hl then SCP_ESP_STORAGE[scpInfo.name] = { Highlight = hl, Object = obj } end
            else
                local data = SCP_ESP_STORAGE[scpInfo.name]
                if not data.Highlight or not data.Highlight.Parent then
                    SCP_ESP_STORAGE[scpInfo.name] = nil
                    local hl = createSCPHighlight(obj, scpInfo.name)
                    if hl then SCP_ESP_STORAGE[scpInfo.name] = { Highlight = hl, Object = obj } end
                end
            end
        else
            if SCP_ESP_STORAGE[scpInfo.name] then
                pcall(function()
                    if SCP_ESP_STORAGE[scpInfo.name].Highlight then
                        SCP_ESP_STORAGE[scpInfo.name].Highlight:Destroy()
                    end
                end)
                SCP_ESP_STORAGE[scpInfo.name] = nil
            end
        end
    end
end

local function getPlayerRelation(plr)
    if not plr or not plr.Team or not player.Team then return Color3.new(1, 1, 1), false end
    local myTeam  = player.Team.Name
    local hisTeam = plr.Team.Name
    if hisTeam == "SCP" or myTeam == "SCP" then
        if hisTeam == myTeam then return Color3.new(0, 1, 0), false end
        return Color3.new(1, 0, 0), true
    end
    local isMeIns  = table.find(TeamGroups.Insurrection, myTeam)
    local isHimIns = table.find(TeamGroups.Insurrection, hisTeam)
    local isEnemy  = false
    if isMeIns and isHimIns then
        isEnemy = false
    elseif not isMeIns and not isHimIns then
        isEnemy = false
    else
        isEnemy = true
    end
    if hisTeam == "Class-D" and not isMeIns then
        local antiRKFolder = workspace:FindFirstChild(plr.Name)
        if not (antiRKFolder and antiRKFolder:FindFirstChild("ANTIRKDISABLED")) then
            isEnemy = false
        end
    end
    return isEnemy and Color3.new(1, 0, 0) or Color3.new(0, 1, 0), isEnemy
end

local GC_CACHE_TTL    = 30
local SCAN_COOLDOWN   = 10
local lastRealScanTime = 0
local unifiedScan = { configs = {}, gunStates = {}, globalsModule = nil, time = 0 }

local function scanGCOnce()
    local now = tick()
    if (now - lastRealScanTime) < SCAN_COOLDOWN then
        return unifiedScan.configs or {}, unifiedScan.gunStates or {}, unifiedScan.globalsModule
    end
    lastRealScanTime = now
    local configs = {}; local gunStates = {}; local gm = nil
    for _, v in ipairs(getgc(true)) do
        if type(v) ~= "table" then continue end
        if type(rawget(v, "Recoil")) == "number"
        and type(rawget(v, "AimRecoil")) == "number"
        and type(rawget(v, "RateOfFire")) == "number" then
            table.insert(configs, v)
        end
        if type(rawget(v, "ammo")) == "number"
        and type(rawget(v, "shooting")) == "boolean"
        and type(rawget(v, "reloading")) == "boolean"
        and rawget(v, "gun") ~= nil
        and typeof(rawget(v, "gun")) == "Instance"
        and rawget(v, "gun"):IsDescendantOf(game) then
            table.insert(gunStates, v)
        end
        if not gm
        and type(rawget(v, "Shoot")) == "function"
        and type(rawget(v, "Reload")) == "function"
        and type(rawget(v, "Raycast")) == "function" then
            gm = v
        end
    end
    unifiedScan.configs = configs; unifiedScan.gunStates = gunStates
    unifiedScan.globalsModule = gm; unifiedScan.time = now
    return configs, gunStates, gm
end

local function forceRescan()
    lastRealScanTime = 0; unifiedScan.time = 0
    return scanGCOnce()
end

local function findGunConfigsCached()
    if unifiedScan.configs and #unifiedScan.configs > 0 and (tick() - unifiedScan.time) < GC_CACHE_TTL then
        return unifiedScan.configs
    end
    if (tick() - lastRealScanTime) < SCAN_COOLDOWN then return unifiedScan.configs or {} end
    scanGCOnce(); return unifiedScan.configs or {}
end

local function findGunStatesCached()
    local now = tick()
    local function filterValid()
        local valid = {}
        for _, gs in ipairs(unifiedScan.gunStates) do
            if rawget(gs, "gun") and typeof(rawget(gs, "gun")) == "Instance" and rawget(gs, "gun"):IsDescendantOf(game) then
                table.insert(valid, gs)
            end
        end
        return valid
    end
    if (now - unifiedScan.time) < GC_CACHE_TTL and #unifiedScan.gunStates > 0 then return filterValid() end
    if (now - lastRealScanTime) < SCAN_COOLDOWN then return filterValid() end
    scanGCOnce(); return filterValid()
end

local function findGlobalsModule()
    if unifiedScan.globalsModule and (tick() - unifiedScan.time) < GC_CACHE_TTL then return unifiedScan.globalsModule end
    if (tick() - lastRealScanTime) < SCAN_COOLDOWN then return unifiedScan.globalsModule end
    scanGCOnce(); return unifiedScan.globalsModule
end

local recoilOriginals = {}

local function patchAllGuns(state)
    noRecoilEnabled = state
    local configs = findGunConfigsCached()
    for _, config in ipairs(configs) do
        if state then
            if not recoilOriginals[config] then
                recoilOriginals[config] = { Recoil = config.Recoil, AimRecoil = config.AimRecoil }
            end
            config.Recoil    = recoilOriginals[config].Recoil    * recoilMultiplier
            config.AimRecoil = recoilOriginals[config].AimRecoil * recoilMultiplier
        else
            if recoilOriginals[config] then
                config.Recoil    = recoilOriginals[config].Recoil
                config.AimRecoil = recoilOriginals[config].AimRecoil
            end
        end
    end

    end

local function pickTargetPart(char)

    for partName, flag in pairs(SILENT_SETTINGS.TargetParts) do
        if flag then table.insert(enabled, partName) end
    end
    if #enabled == 0 then table.insert(enabled, "Head") end
    for i = #enabled, 2, -1 do
        local j = math.random(1, i); enabled[i], enabled[j] = enabled[j], enabled[i]

-- Throttle: пересчитываем цель не каждый кадр, а раз в N кадров
local _closestEnemyFrame  = 0
local _closestEnemyResult = nil
local CLOSEST_ENEMY_INTERVAL = 3  -- пересчёт раз в 3 кадра (~20 раз/сек при 60fps)

local function getClosestEnemyToCursor()
    _closestEnemyFrame = _closestEnemyFrame + 1
    if _closestEnemyFrame < CLOSEST_ENEMY_INTERVAL then
        return _closestEnemyResult
    end
    _closestEnemyFrame = 0

    local camPos   = camera.CFrame.Position
    local mousePos = UserInputService:GetMouseLocation()
    local myChar   = player.Character
    local myRoot   = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local myPos    = myRoot and myRoot.Position or camPos
    -- Кэшируем enabledParts снаружи цикла по игрокам
    local enabledParts = {}
    for partName, flag in pairs(SILENT_SETTINGS.TargetParts) do
        if flag then table.insert(enabledParts, partName) end
    end
    if #enabledParts == 0 then enabledParts = {"Head", "HumanoidRootPart"} end
    local bestTarget = nil; local bestScreenDist = math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == player then continue end
        if not isPlayerAlive(plr) then continue end
        local _, isEnemy = getPlayerRelation(plr)
        if not isEnemy then continue end
        local char = plr.Character
        if not char then continue end
        local bestPartScreenDist = math.huge; local bestPart = nil
        for _, partName in ipairs(enabledParts) do
            local part = char:FindFirstChild(partName)
            if not part then continue end
            local screenPos, onScreen = camera:WorldToScreenPoint(part.Position)
            if not onScreen then continue end
            local sDist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
            if sDist < bestPartScreenDist then bestPartScreenDist = sDist; bestPart = part end
        end
        if not bestPart then
            local fb = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
            if fb then
                local screenPos, onScreen = camera:WorldToScreenPoint(fb.Position)
                if onScreen then
                    bestPartScreenDist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    bestPart = fb
                end
            end
        end
        if not bestPart then continue end
        if bestPartScreenDist > SILENT_SETTINGS.FOV then continue end
        local dist3D = (bestPart.Position - myPos).Magnitude
        local score  = bestPartScreenDist + (dist3D * 0.5)
        if not bestTarget or score < bestTarget.score then
            bestTarget = { plr = plr, score = score }; bestScreenDist = bestPartScreenDist
        end
    end
    _closestEnemyResult = bestTarget and bestTarget.plr or nil
    return _closestEnemyResult
end

local _autoShootCooldown = 0

local function isTargetVisible(targetChar)
    local myChar = player.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot or not targetChar then return false end
    local targetPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart")
    if not targetPart then return false end

    -- Используем оригинальный raycast игры (тот же что в silent aim)
    local rayFn = savedOrig or workspace.Raycast
    local origin = myRoot.Position
    local dir    = targetPart.Position - origin
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { myChar, targetChar }
    params.FilterType = Enum.RaycastFilterType.Exclude

    -- Пробиваем прозрачные/некол объекты как в Globals.Raycast
    local excluded = { myChar, targetChar }
    local TRANSPARENCY_THRESHOLD = 0.5
    local endpoint = targetPart.Position

    for _ = 1, 10 do
        params.FilterDescendantsInstances = excluded
        local result = workspace:Raycast(origin, endpoint - origin, params)
        if result == nil then return true end
        local hit = result.Instance
        if hit.Transparency >= TRANSPARENCY_THRESHOLD or not hit.CanCollide then
            table.insert(excluded, hit)
            origin = result.Position
        else
            return false
        end
    end
    return false
end

-- Heartbeat: auto shoot когда Q зажат + есть цель + цель видна
RunService.Heartbeat:Connect(function()
    if not rapidFireEnabled and not autoShootEnabled then
        return
    end

    local now = tick()

    if rapidFireEnabled then
        if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
            rapidFireLastShot = 0
        else
            local gm = findGlobalsModule()
            if gm and type(gm.Shoot) == "function" then
                local char = player.Character
                if char then
                    local activeState = nil
                    for _, gs in ipairs(findGunStatesCached()) do
                        local gun = rawget(gs, "gun")
                        if gun and gun:IsDescendantOf(char)
                            and type(rawget(gs, "ammo")) == "number" and rawget(gs, "ammo") > 0
                            and not rawget(gs, "reloading")
                            and rawget(gs, "mode") ~= 0 then
                            activeState = gs
                            break
                        end
                    end

                    if activeState then
                        local rof = (activeState.config and activeState.config.RateOfFire) or 600
                        local cooldown = math.max(0.02, (60 / rof) / math.max(1, rapidFireMultiplier))

                        if now - rapidFireLastShot >= cooldown then
                            rapidFireLastShot = now
                            pcall(function()
                                activeState.shooting = false
                                activeState.allowShoot = true
                                gm.Shoot(activeState)
                            end)
                        end
                    end
                end
            end
        end
    end

    if not autoShootEnabled then return end

    local target = cachedSilentTarget
    if not target then return end
    local char = target.Character
    if not char then return end

    local states = findGunStatesCached()
    for _, gs in ipairs(states) do
        if rawget(gs, "reloading") then continue end
        if not rawget(gs, "ammo") or gs.ammo <= 0 then continue end
        if rawget(gs, "allowShoot") == false then continue end
        if rawget(gs, "mode") == 0 then continue end

        -- Считаем cooldown из RateOfFire конфига оружия
        local rof = (gs.config and gs.config.RateOfFire) or 600
        local cooldown = 60 / rof  -- секунд между выстрелами

        if now - _autoShootCooldown < cooldown then return end

        if not isTargetVisible(char) then return end

        local gm = findGlobalsModule()
        if not gm or type(gm.Shoot) ~= "function" then break end

        local targetPart = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
        if not targetPart then break end

        local oldCF   = camera.CFrame
        camera.CFrame = CFrame.new(oldCF.Position, targetPart.Position)

        pcall(gm.Shoot, gs)
        _autoShootCooldown = now

        task.defer(function() camera.CFrame = oldCF end)
        break
    end
end)
local hookFnRef = nil
local hookedGM  = nil   -- сохраняем gm для восстановления
local savedOrig = nil   -- глобальный origRaycast

local function unhookGlobalsModule()
    if hookedGM and savedOrig then
        pcall(function() hookedGM.Raycast = savedOrig end)
    end
    hookedGM  = nil
    savedOrig = nil
    hookFnRef = nil
end

local function hookGlobalsModule()
    unhookGlobalsModule() -- сначала снимаем старый хук если был

    local gm = findGlobalsModule()
    if not gm then return false end

    savedOrig = gm.Raycast
    hookedGM  = gm

    hookFnRef = newcclosure(function(...)
        local args = {...}
        local origin, direction, params, p30
        local isMethodCall = false

        if type(args[1]) == "table" and rawequal(args[1], gm) then
            isMethodCall = true
            origin = args[2]; direction = args[3]; params = args[4]; p30 = args[5]
        else
            origin = args[1]; direction = args[2]; params = args[3]; p30 = args[4]
        end

        if SILENT_SETTINGS.Enabled and cachedSilentTarget then
            local hitRoll = (math.random(1, 100) <= SILENT_SETTINGS.HitChance)

            if hitRoll and cachedSilentTarget.Character then
                local targetPart = pickTargetPart(cachedSilentTarget.Character)
                    or cachedSilentTarget.Character:FindFirstChild("Head")

                if targetPart then
                    local targetPos  = targetPart.Position
                    local newDir     = (targetPos - origin).Unit * direction.Magnitude
                    local finalP30   = p30 or {Transparency = 1, CanCollide = false}

                    gm.Raycast = savedOrig
                    local result
                    if isMethodCall then
                        result = savedOrig(args[1], origin, newDir, params, finalP30)
                    else
                        result = savedOrig(origin, newDir, params, finalP30)
                    end
                    gm.Raycast = hookFnRef
                    return result
                end
            end
        end

        gm.Raycast = savedOrig
        local result = savedOrig(...)
        gm.Raycast = hookFnRef
        return result
    end)

    gm.Raycast = hookFnRef
    return true
end

local function setNoclip(state)
    if state then
        if noclipConnection then
            return
        end

        -- Кэшируем список BasePart'ов при старте и при смене персонажа,
        -- вместо GetDescendants() каждый Stepped
        local cachedParts = {}
        local function addNoclipPart(part)
            if part and part:IsA("BasePart") then
                if noclipOriginalCollisions[part] == nil then
                    noclipOriginalCollisions[part] = part.CanCollide
                end
                table.insert(cachedParts, part)
                part.CanCollide = false
            end
        end

        local function rebuildNoclipCache(char)
            cachedParts = {}
            if char then
                for _, p in ipairs(char:GetDescendants()) do
                    addNoclipPart(p)
                end

                if noclipDescendantConnection then
                    noclipDescendantConnection:Disconnect()
                    noclipDescendantConnection = nil
                end

                noclipDescendantConnection = char.DescendantAdded:Connect(addNoclipPart)
            end
        end
        local char = player.Character
        rebuildNoclipCache(char)
        -- При следующем спауне пересобираем кэш
        noclipCharacterConnection = player.CharacterAdded:Connect(function(newChar)
            rebuildNoclipCache(newChar)
        end)
        noclipConnection = RunService.Stepped:Connect(function()
            for _, part in ipairs(cachedParts) do
                if part and part.Parent then part.CanCollide = false end
            end
        end)
        -- Сохраняем charConn чтобы отключить вместе с noclip
    else
        if noclipConnection then
            noclipConnection:Disconnect()
            noclipConnection = nil
        end

        if noclipCharacterConnection then
            noclipCharacterConnection:Disconnect()
            noclipCharacterConnection = nil
        end

        if noclipDescendantConnection then
            noclipDescendantConnection:Disconnect()
            noclipDescendantConnection = nil
        end

        for part, originalCanCollide in pairs(noclipOriginalCollisions) do
            if part and part.Parent then
                pcall(function()
                    part.CanCollide = originalCanCollide
                end)
            end
        end

        noclipOriginalCollisions = {}
    end
end

local isTeleporting, stopTeleport, flightConn = false, false, nil
local function animatedTeleport(x, y, z)
    if isTeleporting then return end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    isTeleporting = true; stopTeleport = false; teleportingActive = true
    if hum.Sit then hum.Sit = false; task.wait(0.1) end
    hum.PlatformStand = true
    -- Кэшируем части один раз, не пересобираем каждый Stepped
    local noclipParts = {}
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then table.insert(noclipParts, p) end
    end
    flightConn = RunService.Stepped:Connect(function()
        for _, p in ipairs(noclipParts) do
            if p and p.Parent then p.CanCollide = false end
        end
    end)
    local function flyTo(target)
        local reached = false
        local c; c = RunService.Heartbeat:Connect(function(dt)
            if stopTeleport or not root or not root.Parent then reached = true; c:Disconnect(); return end
            root.Velocity = Vector3.new(0, 0, 0)
            local dist = (target - root.Position).Magnitude
            local step = teleportSpeed * dt
            if dist <= step then
                root.CFrame = CFrame.new(target) * root.CFrame.Rotation
                reached = true; c:Disconnect()
            else
                local dir    = (target - root.Position).Unit
                local newPos = root.Position + (dir * step)
                local rot    = (math.abs(dir.Y) > 0.99) and root.CFrame.Rotation or CFrame.lookAt(newPos, newPos + dir).Rotation
                root.CFrame  = CFrame.new(newPos) * rot
            end
        end)
        while not reached do task.wait() end
    end
    local startPos = root.Position
    local highY    = math.max(startPos.Y, y) + teleportHeight
    flyTo(Vector3.new(startPos.X, highY, startPos.Z))
    if not stopTeleport then flyTo(Vector3.new(x, highY, z)) end
    if not stopTeleport then flyTo(Vector3.new(x, y, z)) end
    hum.PlatformStand = false

local function lowerConcat(...)
    local parts = {}
    for i = 1, select("#", ...) do
        local value = select(i, ...)
        if value ~= nil then
            parts[#parts + 1] = tostring(value):lower()
        end
    end
    return table.concat(parts, " ")
end

local function promptMatchesKeywords(prompt, keywords)
    if type(keywords) ~= "table" or #keywords == 0 then
        return true
    end

    local haystack = lowerConcat(prompt.Name, prompt.ObjectText, prompt.ActionText)
    for _, keyword in ipairs(keywords) do
        local needle = tostring(keyword):lower()
        if needle ~= "" and haystack:find(needle, 1, true) then
            return true
        end
    end

    return false
end

local function findPromptNearPosition(targetPos, keywords, maxDistance)
    if not targetPos then
        return nil
    end

    local bestPrompt = nil
    local bestDistance = math.huge
    local searchRadius = maxDistance or 18

    for _, desc in ipairs(workspace:GetDescendants()) do
        if desc:IsA("ProximityPrompt") and promptMatchesKeywords(desc, keywords) then
            local parentPart = desc.Parent and desc.Parent:IsA("BasePart") and desc.Parent or desc:FindFirstAncestorWhichIsA("BasePart")
            if parentPart then
                local distance = (parentPart.Position - targetPos).Magnitude
                if distance <= searchRadius and distance < bestDistance then
                    bestPrompt = desc
                    bestDistance = distance
                end
            end
        end
    end

    return bestPrompt
end

local function neutralizeSCP403Freeze(tool)
    if not tool or not tool.Parent then
        return
    end

    local toolName = tostring(tool.Name or ""):lower()
    if not toolName:find("403", 1, true) and toolName ~= "scp-403" and toolName ~= "scp 403" then
        return
    end

    local changedStates = {}
    local descendantConn = nil

    local function disableDescendant(inst)
        if inst:IsA("Script") or inst:IsA("LocalScript") then
            if changedStates[inst] == nil then
                changedStates[inst] = inst.Disabled
            end
            pcall(function()
                inst.Disabled = true
            end)
        end
    end

    for _, desc in ipairs(tool:GetDescendants()) do
        disableDescendant(desc)
    end

    descendantConn = tool.DescendantAdded:Connect(disableDescendant)

    return function()
        if descendantConn then
            pcall(function()
                descendantConn:Disconnect()
            end)
            descendantConn = nil
        end
        for inst, wasDisabled in pairs(changedStates) do
            if inst and inst.Parent then
                pcall(function()
                    inst.Disabled = wasDisabled
                end)
            end
        end
        changedStates = {}
    end
end

local function lockInstantTpPrompt(prompt)
    if not prompt or not prompt.Parent then
        return false
    end

    local savedEnabled = prompt.Enabled
    local savedHoldDuration = prompt.HoldDuration
    local savedMaxActivationDistance = prompt.MaxActivationDistance
    local savedRequiresLineOfSight = prompt.RequiresLineOfSight

    instantTpPromptRestore = function()
        if prompt and prompt.Parent then
            prompt.Enabled = savedEnabled
            prompt.HoldDuration = savedHoldDuration
            prompt.MaxActivationDistance = savedMaxActivationDistance
            prompt.RequiresLineOfSight = savedRequiresLineOfSight
        end
    end

    prompt.Enabled = false
    return true
end

local function snapshotPromptState(prompt)
    if not prompt or not prompt.Parent then
        return nil
    end

    return {
        Enabled = prompt.Enabled,
        MaxActivationDistance = prompt.MaxActivationDistance,
        HoldDuration = prompt.HoldDuration,
        RequiresLineOfSight = prompt.RequiresLineOfSight,
    }
end

local function restorePromptStateFromSnapshot(prompt, snapshot)
    if not prompt or not prompt.Parent or not snapshot then
        return
    end

    pcall(function() prompt.Enabled = snapshot.Enabled end)
    pcall(function() prompt.MaxActivationDistance = snapshot.MaxActivationDistance end)
    pcall(function() prompt.HoldDuration = snapshot.HoldDuration end)
    pcall(function() prompt.RequiresLineOfSight = snapshot.RequiresLineOfSight end)
end

local function findNamedInstance(root, targetName)
    if not root or not targetName then
        return nil
    end

    local targetLower = string.lower(tostring(targetName))
    if string.lower(root.Name or "") == targetLower then
        return root
    end

    for _, descendant in ipairs(root:GetDescendants()) do
        if string.lower(descendant.Name or "") == targetLower then
            return descendant
        end
    end

    return nil
end

local function triggerPromptNoDistance(prompt, promptState)
    if not prompt or not prompt.Parent then
        return false, "prompt missing"
    end

    local state = promptState or snapshotPromptState(prompt)
    if not state then
        return false, "prompt missing"
    end

    pcall(function() prompt.Enabled = true end)
    pcall(function() prompt.MaxActivationDistance = 9e9 end)
    pcall(function() prompt.HoldDuration = 0 end)
    pcall(function() prompt.RequiresLineOfSight = false end)

    local ok, err
    if type(fireproximityprompt) == "function" then
        ok, err = pcall(fireproximityprompt, prompt)
    else
        ok, err = pcall(function()
            prompt:InputHoldBegin()
            task.wait(0.05)
            prompt:InputHoldEnd()
        end)
    end

    task.delay(0.3, function()
        restorePromptStateFromSnapshot(prompt, state)
    end)

    return ok, err, function()
        restorePromptStateFromSnapshot(prompt, state)
    end
end

local function instantPickupTeleport(config)

    end

        end
    end

        return false
    end

    local title = config.Title or "Instant Tp"
    local maxWait = config.Timeout or PICKUP_RETURN_TIMEOUT
    local returnCFrame = root.CFrame
    local startText = config.StartText or ("Teleporting. Timeout: " .. maxWait .. "s")

    local savedAutoRotate = hum.AutoRotate
    local savedAnchored = root.Anchored
    local shouldStandAfterSnap = config.StandAfterSnap ~= false
    local lockPositionWhileWaiting = config.LockPositionWhileWaiting == true

    teleportingActive = true
    if hum.Sit then hum.Sit = false; task.wait(0.05) end
    hum.PlatformStand = true

    local noclipParts = {}
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            noclipParts[#noclipParts + 1] = p
            noclipOriginalCollisions[p] = p.CanCollide
        end
    end

    local noclipConn = RunService.Stepped:Connect(function()
        for _, p in ipairs(noclipParts) do
            if p then
                p.CanCollide = false
            end
        end
    end)

    local targetCFrame = CFrame.new(targetPos + Vector3.new(0, 2.5, 0))
    root.Anchored = true
    root.CFrame = targetCFrame
    root.Velocity = Vector3.new(0, 0, 0)

    if shouldStandAfterSnap then
        task.delay(0.08, function()
            if instantTpActive and hum and hum.Parent then
                hum.PlatformStand = false
                hum.AutoRotate = true
                pcall(function()
                    hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                end)
            end
        end)
    end

    MIDNIGHT:Notify({ Title = title, Content = startText, Type = "info", Duration = 4 })

    instantTpActive = true
    local startTime = tick()
    local returnQueued = false
    local prompt = resolvedPrompt
    local promptLocked = false
    local backpack = player:FindFirstChild("Backpack")
    local pickupConfirmDelay = math.max(tonumber(config.PickupReturnDelay) or 0.05, 0.05)
    local pickupToken = 0
    local pickupCleanup = nil
    local promptRestoreState = nil
    if prompt and prompt.Parent then
        promptRestoreState = snapshotPromptState(prompt)
    end

    local function restorePromptState()
        restorePromptStateFromSnapshot(prompt, promptRestoreState)
    end

    local function findReturnTool()
        if not returnToolName then
            return nil
        end

        local currentBackpack = player:FindFirstChild("Backpack")
        local currentChar = player.Character

        local found = findNamedInstance(currentBackpack, returnToolName)
        if found then
            return found
        end

        return findNamedInstance(currentChar, returnToolName)
    end

    local function hasReturnTool()
        return findReturnTool() ~= nil
    end

    local function matchesReturnTool(child)
        if not child or not returnToolName then
            return child and child:IsA("Tool") or false
        end

        local targetLower = string.lower(returnToolName)
        local childLower = string.lower(child.Name or "")
        if childLower == targetLower
            or childLower:find(targetLower, 1, true)
            or targetLower:find(childLower, 1, true)
        then
            return true
        end

        return child:FindFirstChild(returnToolName, true) ~= nil
    end

    local function doReturn(reason)
        if not instantTpActive then return end
        instantTpActive = false
        teleportingActive = false
        if pickupCleanup then
            pcall(pickupCleanup)
            pickupCleanup = nil
        end

        if instantTpHeartbeatConn then instantTpHeartbeatConn:Disconnect(); instantTpHeartbeatConn = nil end
        if instantTpBackpackConn then instantTpBackpackConn:Disconnect(); instantTpBackpackConn = nil end
        if instantTpCharacterConn then instantTpCharacterConn:Disconnect(); instantTpCharacterConn = nil end

            Content = reason == "pickup" and successText or timeoutText,
            Type = reason == "pickup" and "success" or "info",

        if not child or returnQueued then
            return
        end

        if not matchesReturnTool(child) then
            return
        end

        pickupToken += 1
        local token = pickupToken

        if type(config.OnPickup) == "function" then
            local ok, cleanup = pcall(config.OnPickup, child)
            if ok and type(cleanup) == "function" then
                if pickupCleanup then
                    pcall(pickupCleanup)
                end
                pickupCleanup = cleanup
            end
        end
        if config.LockPromptOnPickup and not promptLocked then
            promptLocked = lockInstantTpPrompt(prompt)
        end

        task.delay(pickupConfirmDelay, function()
            if not instantTpActive or returnQueued or token ~= pickupToken then
                return
            end
            if not child or not child.Parent then
                return
            end

            local stillInBackpack = backpack and child:IsDescendantOf(backpack)
            local stillInCharacter = char and child:IsDescendantOf(char)
            if stillInBackpack or stillInCharacter then
                returnQueued = true
                doReturn("pickup")
            end
        end)
    end

    if backpack then
        instantTpBackpackConn = backpack.ChildAdded:Connect(function(child)
            schedulePickupReturn(child)
        end)
    end

    if char then
        instantTpCharacterConn = char.ChildAdded:Connect(function(child)
            schedulePickupReturn(child)
        end)
    end

    instantTpHeartbeatConn = RunService.Heartbeat:Connect(function()
        if not instantTpActive then return end
        if tick() - startTime > maxWait then
            doReturn("timeout")
            return
        end
        if hasReturnTool() and not returnQueued then
            returnQueued = true
            doReturn("pickup")
            return
        end
        if lockPositionWhileWaiting and root and root.Parent then
            root.Velocity = Vector3.new(0, 0, 0)
            root.CFrame = targetCFrame
        end
    end)

    task.spawn(function()
        while instantTpActive do
            task.wait(0.1)
        end
    end)

    return true
end

local function freezeTeleport(x, y, z, options)
    options = options or {}
    if freezeLockActive then
        MIDNIGHT:Notify({ Title = "Fast TP", Content = "Already active. Wait for the return.", Type = "info", Duration = 4 })
        return
    end

    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    local returnCFrame = root.CFrame
    local rootOriginalAnchored = root.Anchored
    local maxWait = options.Timeout or PICKUP_RETURN_TIMEOUT
    local pickupConfirmDelay = math.max(tonumber(options.PickupReturnDelay) or 0.15, 0.05)
    local onPickup = options.OnPickup
    local disableNoclip = options.DisableNoclip == true
    local trackPickupReturn = options.DisablePickupReturn ~= true
    teleportingActive = true
    if hum.Sit then hum.Sit = false; task.wait(0.05) end
    hum.PlatformStand = true
    if options.AnchorRoot == true then
        pcall(function()
            root.Anchored = true
        end)
    end

    local noclipParts = {}
    if not disableNoclip then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then
                noclipParts[#noclipParts + 1] = p
                noclipOriginalCollisions[p] = p.CanCollide
            end
        end
    end

    local noclipConn = nil
    if not disableNoclip then
        noclipConn = RunService.Stepped:Connect(function()
            for _, p in ipairs(noclipParts) do
                if p and p.Parent then
                    p.CanCollide = false
                end
            end
        end)
    end

    local targetPos = CFrame.new(x, y, z)
    root.CFrame = targetPos
    root.Velocity = Vector3.new(0, 0, 0)
    local startText = options.StartText or ("Teleporting. Pick up the weapon - auto return. Timeout: " .. maxWait .. "s")
    local successText = options.SuccessText or "Weapon picked up. Returning."
    local timeoutText = options.TimeoutText or "Timeout. Returning."
    local title = options.Title or "Fast TP"
    local returnToolName = options.ReturnToolName
    local clickDetector = options.ClickDetector
    local promptToSpam = options.Prompt
    local spamAction = options.SpamAction
    local triggerInterval = math.max(tonumber(options.TriggerInterval) or 0.05, 0.03)
    local spamStartDelay = math.max(tonumber(options.SpamStartDelay) or 0, 0)
    local spamImmediate = options.SpamImmediate == true
    local promptRestoreState = nil
    if promptToSpam and promptToSpam.Parent then
        promptRestoreState = snapshotPromptState(promptToSpam)
    end

    local function restorePromptState()
        restorePromptStateFromSnapshot(promptToSpam, promptRestoreState)
    end
    local function doSpamAttempt()
        if type(spamAction) == "function" then
            pcall(spamAction)
        elseif clickDetector and clickDetector.Parent and type(fireclickdetector) == "function" then
            pcall(fireclickdetector, clickDetector)
        elseif promptToSpam and promptToSpam.Parent then
            pcall(triggerPromptNoDistance, promptToSpam, promptRestoreState)
        end
    end
    MIDNIGHT:Notify({ Title = title, Content = startText, Type = "info", Duration = 4 })

    freezeLockActive = true
    local startTime = tick()
    local pickupEligibleAt = startTime + math.max(tonumber(options.PickupWatchDelay) or 0, 0)
    local weaponReturnQueued = false
    local charWeaponConn = nil
    local backpack = player:FindFirstChild("Backpack")
    local pickupCleanup = nil
    local noclipOriginalCollisions = {}

    local function matchesReturnTool(child)
        if not child or not returnToolName then
            return child and child:IsA("Tool") or false
        end

        local targetLower = string.lower(returnToolName)
        local childLower = string.lower(child.Name or "")
        if childLower == targetLower
            or childLower:find(targetLower, 1, true)
            or targetLower:find(childLower, 1, true)
        then
            return true
        end

        return child:FindFirstChild(returnToolName, true) ~= nil
    end

    local function findReturnTool()
        if not returnToolName then
            return nil
        end

        local currentBackpack = player:FindFirstChild("Backpack")
        local currentChar = player.Character

        local found = findNamedInstance(currentBackpack, returnToolName)
        if found then
            return found
        end

        return findNamedInstance(currentChar, returnToolName)
    end

    local function hasReturnTool()
        return findReturnTool() ~= nil
    end

    local function doReturn(reason)
        if not freezeLockActive then return end
        freezeLockActive = false
        teleportingActive = false
        if pickupCleanup then
            pcall(pickupCleanup)
            pickupCleanup = nil
        end
        if freezeLockConn then freezeLockConn:Disconnect(); freezeLockConn = nil end
        if freezeWeaponConn then freezeWeaponConn:Disconnect(); freezeWeaponConn = nil end
        if charWeaponConn then charWeaponConn:Disconnect(); charWeaponConn = nil end
        if root and root.Parent then
            pcall(function()
                root.Anchored = rootOriginalAnchored
            end)
            root.CFrame = returnCFrame
            root.Velocity = Vector3.new(0, 0, 0)
        end
        hum.PlatformStand = false
        if noclipConn then noclipConn:Disconnect() end
        for part, originalCanCollide in pairs(noclipOriginalCollisions) do
            if part and part.Parent then
                pcall(function()
                    part.CanCollide = originalCanCollide
                end)
            end
        end
        noclipOriginalCollisions = {}
        restorePromptState()
        task.delay(0.15, restorePromptState)
        MIDNIGHT:Notify({
            Title = title,
            Content = reason == "weapon" and successText or timeoutText,
            Type = "success",
            Duration = 3,
        })
    end

    if spamImmediate then
        doSpamAttempt()
    end

    task.spawn(function()
        while freezeLockActive do
            if tick() - startTime >= spamStartDelay then
                doSpamAttempt()
            end
            task.wait(triggerInterval)
        end
    end)

    if freezeWeaponConn then freezeWeaponConn:Disconnect(); freezeWeaponConn = nil end
    if trackPickupReturn and backpack then
        freezeWeaponConn = backpack.ChildAdded:Connect(function(child)
            if tick() < pickupEligibleAt then
                return
            end
            if matchesReturnTool(child) and not weaponReturnQueued then
                weaponReturnQueued = true
                if type(onPickup) == "function" then
                    local ok, cleanup = pcall(onPickup, child)
                    if ok and type(cleanup) == "function" then
                        if pickupCleanup then
                            pcall(pickupCleanup)
                        end
                        pickupCleanup = cleanup
                    end
                end
                task.wait(pickupConfirmDelay)
                doReturn("weapon")
            end
        end)
    end

    if trackPickupReturn and char then
        charWeaponConn = char.ChildAdded:Connect(function(child)
            if tick() < pickupEligibleAt then
                return
            end
            if matchesReturnTool(child) and not weaponReturnQueued then
                weaponReturnQueued = true
                if type(onPickup) == "function" then
                    local ok, cleanup = pcall(onPickup, child)
                    if ok and type(cleanup) == "function" then
                        if pickupCleanup then
                            pcall(pickupCleanup)
                        end
                        pickupCleanup = cleanup
                    end
                end
                task.wait(pickupConfirmDelay)
                doReturn("weapon")
                if charWeaponConn then charWeaponConn:Disconnect(); charWeaponConn = nil end
            end
        end)
    end

    if freezeLockConn then freezeLockConn:Disconnect(); freezeLockConn = nil end
    freezeLockConn = RunService.Heartbeat:Connect(function()
        if not freezeLockActive then
            if charWeaponConn then charWeaponConn:Disconnect(); charWeaponConn = nil end
            return
        end
        if tick() - startTime > maxWait then
            if charWeaponConn then charWeaponConn:Disconnect(); charWeaponConn = nil end
            doReturn("timeout")
            return
        end
        if trackPickupReturn and tick() >= pickupEligibleAt and hasReturnTool() and not weaponReturnQueued then
            weaponReturnQueued = true
            doReturn("weapon")
            return
        end
        if root and root.Parent then
            root.Velocity = Vector3.new(0, 0, 0)
            root.CFrame = targetPos
        end
    end)
end

    local bestDistance = math.huge

    for _, desc in ipairs(workspace:GetDescendants()) do
        if desc:IsA("ProximityPrompt") and desc.ObjectText == "Bandage" then
            local parentPart = desc.Parent and desc.Parent:IsA("BasePart") and desc.Parent or desc:FindFirstAncestorWhichIsA("BasePart")
            if parentPart and root then
                local distance = (root.Position - parentPart.Position).Magnitude
                if distance < bestDistance then
                    bestDistance = distance
                    bestPrompt = desc
                end
            elseif not bestPrompt then
                bestPrompt = desc
            end
        end
    end

    return bestPrompt
end

local function getBandageTargetPosition(prompt)
    if not prompt then return nil end
    local parent = prompt.Parent
    if parent and parent:IsA("BasePart") then
        return parent.Position
    end

    local part = prompt:FindFirstAncestorWhichIsA("BasePart")
    if part then
        return part.Position
    end

    local model = prompt:FindFirstAncestorWhichIsA("Model")
    if model then
        if model.PrimaryPart then
            return model.PrimaryPart.Position
        end
        local anyPart = model:FindFirstChildWhichIsA("BasePart", true)
        if anyPart then
            return anyPart.Position
        end
        local ok, pivot = pcall(function() return model:GetPivot() end)
        if ok and pivot then
            return pivot.Position
        end
    end

    return nil
end

local function resolveBandageTeleportTarget()
    local prompt = findBandagePrompt()
    if not prompt then
        return nil
    end

    return getBandageTargetPosition(prompt), prompt
end

local function findMedstationId()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then
        return nil
    end

    local closestId = nil
    local closestDist = math.huge

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and string.find(string.lower(obj.Name), "medstation", 1, true) then
            local networkId = tonumber(obj:GetAttribute("NetworkId"))
            if networkId then
                local root = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                if root then
                    local dist = (root.Position - hrp.Position).Magnitude
                    if dist < closestDist then
                        closestDist = dist
                        closestId = networkId
                    end
                end
            end
        end
    end

    return closestId
end

local function useMedstation(networkId)
    if not networkId then
        return false, "medstation id missing"
    end

    local ok, err = pcall(function()
        local buf = buffer.create(6)
        buffer.writeu8(buf, 0, 49)
        buffer.writeu32(buf, 1, networkId)
        buffer.writeu8(buf, 5, 0)
        PacketObject:FireServer(buf)
    end)

    return ok, err
end

local function bandageTeleport()
    if freezeLockActive or bandageLockActive then
        MIDNIGHT:Notify({ Title = "Bandage Taker", Content = "Already active. Wait for return.", Type = "info", Duration = 4 })
        return
    end

    local char = player.Character
    local maxWait = PICKUP_RETURN_TIMEOUT
    MIDNIGHT:Notify({ Title = "Bandage Taker", Content = "Sending medstation packet. Timeout: " .. maxWait .. "s", Type = "info", Duration = 4 })

    bandageLockActive = true
    local startTime = tick()
    local bandageReturnQueued = false
    local bandageSpamInterval = 0.05

    local function doReturn(reason)
        if not bandageLockActive then return end
        bandageLockActive = false
        if bandageLockConn then bandageLockConn:Disconnect(); bandageLockConn = nil end
        if bandageToolConn then bandageToolConn:Disconnect(); bandageToolConn = nil end
        if bandageCharConn then bandageCharConn:Disconnect(); bandageCharConn = nil end
        MIDNIGHT:Notify({
            Title = "Bandage Taker",
            Content = reason == "pickup" and "Bandage picked up." or "Timeout.",
            Type = reason == "pickup" and "success" or "info",
            Duration = 3
        })
    end

    if bandageToolConn then bandageToolConn:Disconnect(); bandageToolConn = nil end
    local backpack = player:FindFirstChild("Backpack")
    if backpack then
        bandageToolConn = backpack.ChildAdded:Connect(function(child)
            local childName = string.lower(child.Name)
            if child:IsA("Tool") and childName:find("bandage", 1, true) and not bandageReturnQueued then
                bandageReturnQueued = true
                doReturn("pickup")
            end
        end)
    end

    if char then
        bandageCharConn = char.ChildAdded:Connect(function(child)
            local childName = string.lower(child.Name)
            if child:IsA("Tool") and childName:find("bandage", 1, true) and not bandageReturnQueued then
                bandageReturnQueued = true
                doReturn("pickup")
            end
        end)
    end

    if bandageLockConn then bandageLockConn:Disconnect(); bandageLockConn = nil end

    task.spawn(function()
        while bandageLockActive do
            if tick() - startTime > maxWait then
                doReturn("timeout")
                break
            end
            local medstationId = findMedstationId()
            if medstationId then
                local ok = useMedstation(medstationId)
                if not ok then
                    doReturn("timeout")
                    break
                end
            end
            task.wait(bandageSpamInterval)
        end
    end)
end

local function resolveSCP096AnchorCFrame()
    local scp096 = workspace:FindFirstChild("SCP096")
    if not scp096 then
        return nil
    end

    local humanoid = scp096:FindFirstChildOfClass("Humanoid")
    if humanoid and humanoid.RootPart and humanoid.RootPart:IsA("BasePart") then
        return humanoid.RootPart.CFrame
    end

    local preferredNames = {
        "HumanoidRootPart",
        "Head",
        "UpperTorso",
        "Torso",
        "LowerTorso",
        "RootPart",
    }

    for _, name in ipairs(preferredNames) do
        local part = scp096:FindFirstChild(name, true)
        if part and part:IsA("BasePart") then
            return part.CFrame
        end
    end

    local model = scp096:IsA("Model") and scp096 or scp096:FindFirstChildWhichIsA("Model", true)
    if model then
        local modelPart = model.PrimaryPart
            or model:FindFirstChild("HumanoidRootPart", true)
            or model:FindFirstChild("Head", true)
            or model:FindFirstChild("UpperTorso", true)
            or model:FindFirstChild("Torso", true)
            or model:FindFirstChild("LowerTorso", true)
        if modelPart and modelPart:IsA("BasePart") then
            return modelPart.CFrame
        end

        local ok, pivot = pcall(function()
            return model:GetPivot()
        end)
        if ok and pivot then
            return pivot
        end
    end

    local anyBasePart = scp096:FindFirstChildWhichIsA("BasePart", true)
    if anyBasePart then
        return anyBasePart.CFrame
    end

    return nil
end

local SCP096_STAGE_CFRAME = CFrame.new(
    Vector3.new(1546.95, -202.15, 1023.96),
    Vector3.new(1546.01, -202.48, 1023.95)
)

local function resolveSCP096FrontCFrame()
    local anchorCFrame = resolveSCP096AnchorCFrame()
    if not anchorCFrame then
        return nil
    end

    local faceDistance = 2.5
    local lookVector = anchorCFrame.LookVector
    local flatLook = Vector3.new(lookVector.X, 0, lookVector.Z)
    if flatLook.Magnitude < 0.05 then
        flatLook = Vector3.new(0, 0, -1)
    else
        flatLook = flatLook.Unit
    end

    local targetPosition = anchorCFrame.Position + flatLook * faceDistance
    targetPosition = Vector3.new(targetPosition.X, anchorCFrame.Position.Y - 1.5, targetPosition.Z)

    return CFrame.lookAt(targetPosition, Vector3.new(anchorCFrame.Position.X, targetPosition.Y, anchorCFrame.Position.Z))
end

local function removeSCP096Freeze()
    local scp096 = workspace:FindFirstChild("SCP096")
    local main096 = scp096 and scp096:FindFirstChild("Main096")
    local freezeScript = main096 and main096:FindFirstChild("c")

    if freezeScript and (freezeScript:IsA("Script") or freezeScript:IsA("LocalScript")) then
        pcall(function()
            freezeScript.Disabled = true
        end)
        pcall(function()
            freezeScript:Destroy()
        end)
    end
end

local function teleportToSCP096Fast()
    if scp096FastTpActive then
        MIDNIGHT:Notify({ Title = "Fast TP", Content = "Already active. Wait for the return.", Type = "info", Duration = 4 })
        return false
    end

    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then
        return false
    end

    local stageCFrame = resolveSCP096TeleportTarget()
    if typeof(stageCFrame) ~= "CFrame" then
        MIDNIGHT:Notify({ Title = "Fast TP", Content = "SCP-096 stage point not found.", Type = "error", Duration = 3 })
        return false
    end

    local returnCFrame = root.CFrame
    local savedAutoRotate = hum.AutoRotate
    local savedAnchored = root.Anchored
    local stageWaitTime = PICKUP_RETURN_TIMEOUT
    local faceHoldTime = 10
    local freezeSpamDelay = math.max(faceHoldTime - 2, 0)
    local maxWait = stageWaitTime
    local activeTargetCFrame = stageCFrame

    removeSCP096Freeze()

    scp096FastTpActive = true
    teleportingActive = true
    if hum.Sit then
        hum.Sit = false
        task.wait(0.05)
    end
    hum.PlatformStand = true
    hum.AutoRotate = false
    root.Anchored = true

    local noclipParts = {}
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            noclipParts[#noclipParts + 1] = part
        end
    end

    local noclipConn = RunService.Stepped:Connect(function()
        for _, part in ipairs(noclipParts) do
            if part and part.Parent then
                part.CanCollide = false
            end
        end
    end)

    root.CFrame = activeTargetCFrame
    root.Velocity = Vector3.new(0, 0, 0)

    MIDNIGHT:Notify({
        Title = "Fast TP",
        Content = "Stage first, then SCP-096 face, then return. Stage timeout: " .. stageWaitTime .. "s, face hold: " .. faceHoldTime .. "s",
        Type = "info",
        Duration = 4,
    })

    local startTime = tick()

    local function doReturn(reason)
        if not scp096FastTpActive then return end
        scp096FastTpActive = false
        teleportingActive = false

        if scp096FastTpConn then
            scp096FastTpConn:Disconnect()
            scp096FastTpConn = nil
        end

        if root and root.Parent then
            root.CFrame = returnCFrame
            root.Velocity = Vector3.new(0, 0, 0)
            root.Anchored = savedAnchored
        end

        if hum and hum.Parent then
            hum.PlatformStand = false
            hum.AutoRotate = savedAutoRotate
        end

        if noclipConn then
            noclipConn:Disconnect()
        end

        MIDNIGHT:Notify({
            Title = "Fast TP",
            Content = reason == "timeout" and "Timeout. Returning." or "Aggroed. Returning.",
            Type = reason == "timeout" and "info" or "success",
            Duration = 3,
        })
    end

    if scp096FastTpConn then
        scp096FastTpConn:Disconnect()
        scp096FastTpConn = nil
    end
    scp096FastTpConn = RunService.Heartbeat:Connect(function()
        if not scp096FastTpActive then return end
        if tick() - startTime > maxWait then
            doReturn("timeout")
            return
        end
        if root and root.Parent then
            root.Velocity = Vector3.new(0, 0, 0)
            root.CFrame = activeTargetCFrame
        end
    end)

    task.spawn(function()
        task.wait(0.1)
        while scp096FastTpActive do
            if tick() - startTime > maxWait then
                doReturn("timeout")
                return
            end

            removeSCP096Freeze()
            local frontCFrame = resolveSCP096FrontCFrame()
            if frontCFrame then
                startTime = tick()
                maxWait = faceHoldTime + 0.5
                removeSCP096Freeze()
                activeTargetCFrame = frontCFrame
                if root and root.Parent then
                    root.CFrame = activeTargetCFrame
                    root.Velocity = Vector3.new(0, 0, 0)
                    root.Anchored = false
                end
                if hum and hum.Parent then
                    hum.PlatformStand = false
                    hum.AutoRotate = savedAutoRotate
                end

                local faceStartTime = tick()
                task.spawn(function()
                    task.wait(freezeSpamDelay)
                    while scp096FastTpActive and (tick() - faceStartTime) < faceHoldTime do
                        removeSCP096Freeze()
                        task.wait(0.05)
                    end
                end)

                task.delay(faceHoldTime, function()
                    if scp096FastTpActive then
                        if root and root.Parent then
                            root.Anchored = false
                        end
                        doReturn("agro")
                    end
                end)
                return
            end

            task.wait(0.05)
        end
    end)

    return true
end

local function teleportToSCP096Safe()
    return teleportToSCP096Fast()
end

local PUMP_PATHS = {
    Red    = {"InteriorBuild", "Interactables", "GeneratorSystem", "ControlRed",    "Button", "ProximityPrompt"},
    Yellow = {"InteriorBuild", "Interactables", "GeneratorSystem", "ControlYellow", "Button", "ProximityPrompt"},
}

local pumpTeleportActive = false
local pumpTeleportHeartbeatConn = nil
local pumpPromptRestore = nil
local PUMP_RETURN_TIMEOUT = 5
PUMP_SPAM_DELAY = 0.08

function findByPath(pathTable)
    local current = workspace
    for _, part in ipairs(pathTable) do
        if not current then return nil end
        current = current:FindFirstChild(part)
    end
    return current
end

function findPumpPromptFallback(controlName)
    for _, desc in ipairs(workspace:GetDescendants()) do
        local name = string.lower(desc.Name)
        local colorName = controlName == "ControlRed" and "red" or "yellow"
        local exactMatch = desc.Name == controlName
        local looseMatch = name:find("control", 1, true) and name:find(colorName, 1, true)
        if exactMatch or looseMatch then
            local btn = desc:FindFirstChild("Button", true)
            if btn then
                local pp = btn:FindFirstChildWhichIsA("ProximityPrompt", true)
                if pp then return pp end
            end
            local pp = desc:FindFirstChildWhichIsA("ProximityPrompt", true)
            if pp then return pp end
        end
    end
    return nil
end

function getPromptTargetPosition(prompt)
    if not prompt then return nil end
    local parent = prompt.Parent
    if parent and parent:IsA("BasePart") then
        return parent.Position
    end
    local part = prompt:FindFirstAncestorWhichIsA("BasePart")
    if part then
        return part.Position
    end
    local model = prompt:FindFirstAncestorWhichIsA("Model")
    if model then
        if model.PrimaryPart then
            return model.PrimaryPart.Position
        end
        local anyPart = model:FindFirstChildWhichIsA("BasePart", true)
        if anyPart then
            return anyPart.Position
        end
        local ok, pivot = pcall(function() return model:GetPivot() end)
        if ok and pivot then
            return pivot.Position
        end
    end
    return nil
end

function resolvePumpPrompt(colorKey)
    local prompt = findByPath(PUMP_PATHS[colorKey])
    local controlName = colorKey == "Red" and "ControlRed" or "ControlYellow"

    if prompt and prompt:IsA("ProximityPrompt") then
        return prompt, getPromptTargetPosition(prompt)
    end

    local control = workspace:FindFirstChild(controlName, true)
    if control then
        local btn = control:FindFirstChild("Button", true)
        if btn then
            local pp = btn:FindFirstChildWhichIsA("ProximityPrompt", true)
            if pp then
                return pp, getPromptTargetPosition(pp)
            end
        end
        local pp = control:FindFirstChildWhichIsA("ProximityPrompt", true)
        if pp then
            return pp, getPromptTargetPosition(pp)
        end
    end

    prompt = findPumpPromptFallback(controlName)
    return prompt, getPromptTargetPosition(prompt)
end

function restorePumpPrompt()
    if pumpPromptRestore then
        pcall(pumpPromptRestore)
        pumpPromptRestore = nil
    end
end

function instantPumpTeleport(colorKey)
    if pumpTeleportActive then
        MIDNIGHT:Notify({ Title = "Pump " .. colorKey, Content = "Already active. Wait for return.", Type = "info", Duration = 4 })
        return false
    end

    local prompt, targetPos = resolvePumpPrompt(colorKey)
    if not prompt or not targetPos then
        MIDNIGHT:Notify({ Title = "Pump " .. colorKey, Content = "ProximityPrompt not found!", Type = "error", Duration = 3 })
        return false
    end

    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return false end

    local returnCFrame = root.CFrame
    local title = "Pump " .. colorKey
    local startActionText = prompt.ActionText or ""
    local maxWait = PUMP_RETURN_TIMEOUT
    local startTime = tick()
    local targetCFrame = CFrame.new(targetPos + Vector3.new(0, 2.5, 0))
    local pumpPromptState = snapshotPromptState(prompt)

    pumpTeleportActive = true
    teleportingActive = true
    if hum.Sit then hum.Sit = false; task.wait(0.05) end
    hum.PlatformStand = true

    local noclipParts = {}
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            noclipParts[#noclipParts + 1] = p
        end
    end

    local noclipConn = RunService.Stepped:Connect(function()
        for _, p in ipairs(noclipParts) do
            if p and p.Parent then
                p.CanCollide = false
            end
        end
    end)

    root.CFrame = targetCFrame
    root.Velocity = Vector3.new(0, 0, 0)

    MIDNIGHT:Notify({
        Title = title,
        Content = "Teleporting. Spamming F until pump switches. Timeout: " .. maxWait .. "s",
        Type = "info",
        Duration = 4,
    })

    local function doReturn(reason)
        if not pumpTeleportActive then return end
        pumpTeleportActive = false
        teleportingActive = false
        restorePumpPrompt()

        if pumpTeleportHeartbeatConn then pumpTeleportHeartbeatConn:Disconnect(); pumpTeleportHeartbeatConn = nil end

        if root and root.Parent then
            root.CFrame = returnCFrame
            root.Velocity = Vector3.new(0, 0, 0)
        end
        hum.PlatformStand = false
        if noclipConn then noclipConn:Disconnect() end

        MIDNIGHT:Notify({
            Title = title,
            Content = reason == "switch" and "Pump switched. Returning." or "Timeout. Returning.",
            Type = reason == "switch" and "success" or "info",
            Duration = 3,
        })
    end

    pumpTeleportHeartbeatConn = RunService.Heartbeat:Connect(function()
        if not pumpTeleportActive then return end
        if tick() - startTime > maxWait then
            doReturn("timeout")
            return
        end
        if root and root.Parent then
            root.Velocity = Vector3.new(0, 0, 0)
            root.CFrame = targetCFrame
        end
    end)

    task.spawn(function()
        while pumpTeleportActive do
            if not prompt or not prompt.Parent then
                prompt, targetPos = resolvePumpPrompt(colorKey)
                if prompt and prompt.Parent and targetPos then
                    targetCFrame = CFrame.new(targetPos + Vector3.new(0, 2.5, 0))
                    pumpPromptState = snapshotPromptState(prompt)
                end
            end

            if prompt and prompt.Parent then
                local ok, err, restorePrompt = triggerPromptNoDistance(prompt, pumpPromptState)
                if type(restorePrompt) == "function" then
                    restorePumpPrompt()
                    pumpPromptRestore = restorePrompt
                end

                if (prompt.ActionText or "") ~= startActionText or not prompt.Enabled then
                    doReturn("switch")
                    return
                end
            end

            if VirtualInputManager then
                pcall(function()
                    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F, false, game)
                    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game)
                end)
            end

            task.wait(PUMP_SPAM_DELAY)
        end
    end)

    return true
end

end

rebuildESP = rebuildESP or function(plr)
    cleanESP(plr)
    if plr == player then return end
    local char = plr.Character
    if not char then return nil end
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not ESP_SETTINGS.Enabled or not root or not hum then return nil end
    if not plr.Team or not player.Team then return nil end
    local _, isEnemy = getPlayerRelation(plr)
    if not isEnemy then return false end
    ESP_STORAGE[plr] = {}
    local storage = ESP_STORAGE[plr]
    if ESP_SETTINGS.Chams then
        storage.Highlight = Instance.new("Highlight", char)
        storage.Highlight.FillTransparency = 0.85
        local teamCol = getTeamColor3(plr)
        storage.Highlight.FillColor = teamCol; storage.Highlight.OutlineColor = teamCol
    end
    local showFull = ESP_SETTINGS.Names; local showHP = ESP_SETTINGS.Health
    if showFull or showHP then
        local built = buildGui(root, showFull)
        for k, v in pairs(built) do storage[k] = v end
    end
    return true
end

-- Кэш последних значений HP на игрока — не пересчитываем цвет если не изменилось
espHpCache = {}  -- [plr] = { hp = N, maxHp = N, color = Color3 }

espUpdateCounter = 0
espValidateCounter = 0
ESP_UPDATE_INTERVAL   = 3   -- обновляем GUI каждые 3 Heartbeat (~20fps)
ESP_VALIDATE_INTERVAL = 90  -- проверяем наличие ESP раз в 90 тиков (~1.5сек)

updateESPData = updateESPData or function()
    if not ESP_SETTINGS.Enabled then
        return
    end

    espUpdateCounter = espUpdateCounter + 1
    local doUpdate = (espUpdateCounter >= ESP_UPDATE_INTERVAL)
    if doUpdate then espUpdateCounter = 0 end

    espValidateCounter = espValidateCounter + 1
    local doValidate = (espValidateCounter >= ESP_VALIDATE_INTERVAL)
    if doValidate then espValidateCounter = 0 end

    -- Лёгкая проверка живости инстансов — без pcall, через rawequal на Parent
    for _, plr in ipairs(Players:GetPlayers()) do
        local storage = ESP_STORAGE[plr]
        if not storage then continue end

        -- Проверка живости Gui (дёшево, без pcall)
        if storage.Gui then
            if not storage.Gui.Parent then cleanESP(plr); espHpCache[plr] = nil; continue end
        end
        -- Проверка живости Highlight
        if storage.Highlight and not storage.Highlight.Parent then
            storage.Highlight = nil
        end

        if not doUpdate then continue end  -- GUI-обновление только раз в N тиков

        local char = plr.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if not hum then continue end

        local relColor, isEnemy = getPlayerRelation(plr)

        -- Highlight: пересоздаём только при смене врага/друга
        if storage.Highlight then
            if not isEnemy then
                storage.Highlight:Destroy(); storage.Highlight = nil
            else
                local teamCol = getTeamColor3(plr)
                storage.Highlight.OutlineColor = teamCol
                storage.Highlight.FillColor    = teamCol
            end
        elseif ESP_SETTINGS.Chams and isEnemy and char then
            storage.Highlight = Instance.new("Highlight", char)
            storage.Highlight.FillTransparency = 0.85
            local teamCol = getTeamColor3(plr)
            storage.Highlight.FillColor    = teamCol
            storage.Highlight.OutlineColor = teamCol
        end

        if storage.Gui then
            if storage.BgStroke then storage.BgStroke.Color = relColor end

            -- HP: Color3.fromHSV и string.format только при реальном изменении здоровья
            local curHp    = math.floor(hum.Health)
            local curMaxHp = math.floor(hum.MaxHealth)
            local cache    = espHpCache[plr]
            local hpDirty  = not cache or cache.hp ~= curHp or cache.maxHp ~= curMaxHp
            local hpP, hpColor
            if hpDirty then
                Circle = UDim2.new(0, 140, 0, 58) -- Just safety logic
                hpP    = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                hpColor = Color3.fromHSV(hpP * 0.33, 1, 1)
                espHpCache[plr] = { hp = curHp, maxHp = curMaxHp, hpP = hpP, color = hpColor }
            else
                hpP    = cache.hpP
                hpColor = cache.color
            end

            if storage.mode == "full" then
                local teamName = (plr.Team and plr.Team.Name) or "No Team"
                if storage.Dot       then storage.Dot.BackgroundColor3 = relColor end
                if storage.NameLabel then storage.NameLabel.Visible = ESP_SETTINGS.Names; storage.NameLabel.Text = plr.Name end
                if storage.TeamLabel then storage.TeamLabel.Visible = ESP_SETTINGS.Names; storage.TeamLabel.Text = teamName; storage.TeamLabel.TextColor3 = relColor end
                if storage.HBG       then storage.HBG.Visible = ESP_SETTINGS.Health end
                if storage.HTLabel   then storage.HTLabel.Visible = ESP_SETTINGS.Health end
                if ESP_SETTINGS.Health and hpDirty then
                    if storage.HMain   then storage.HMain.Size = UDim2.new(hpP, 0, 1, 0); storage.HMain.BackgroundColor3 = hpColor end
                    if storage.HTLabel then storage.HTLabel.Text = string.format("%d/%d HP", curHp, curMaxHp); storage.HTLabel.TextColor3 = hpColor end
                end
            elseif storage.mode == "mini" and hpDirty then
                if storage.HMain   then storage.HMain.Size = UDim2.new(hpP, 0, 1, 0); storage.HMain.BackgroundColor3 = hpColor end
                if storage.HTLabel then storage.HTLabel.Text = string.format("%d/%d", curHp, curMaxHp); storage.HTLabel.TextColor3 = hpColor end
            end
        end
    end

    -- Периодическая валидация: кто должен иметь ESP, а кто нет
    if doValidate then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr == player then continue end
            local _, isEnemy = getPlayerRelation(plr)
            local hasESP = ESP_STORAGE[plr] ~= nil
            if isEnemy and not hasESP then rebuildESP(plr)
            elseif not isEnemy and hasESP then cleanESP(plr); espHpCache[plr] = nil end
        end
    end
end

function getStaffRankColor(rank)
    rank = tonumber(rank) or 0
    if rank >= 200 then
        return Color3.fromRGB(255, 96, 96)
    elseif rank >= 100 then
        return Color3.fromRGB(255, 196, 92)
    elseif rank >= 50 then
        return Color3.fromRGB(120, 220, 140)
    end
    return Color3.fromRGB(96, 190, 255)
end

function updateAdminPresence(plr, roleName, rank, actionText, commandText, options)
    if not adminPresenceWd or not plr then return end
    pcall(adminPresenceWd.TrackAdmin, adminPresenceWd, plr, tostring(roleName or "Admin"), getStaffRankColor(rank), actionText, commandText, options)
end

function removeAdminPresence(plr)
    if not adminPresenceWd or not plr then return end
    pcall(adminPresenceWd.RemoveAdmin, adminPresenceWd, plr)
end

function handleAdminChatMessage(plr, roleName, msg)
    if not plr or not msg then return end
    local sender = msg.TextSource and msg.TextSource.Name or ""
    if sender ~= plr.Name then return end
    if addAdminLog then
        addAdminLog(string.format("[%s] %s said: %s", roleName, plr.Name, msg.Text or ""), Color3.fromRGB(255, 200, 80), "chat", true)
    end
    local adminInfo = onlineAdmins[plr.UserId]
    updateAdminPresence(plr, roleName, adminInfo and adminInfo.rank, "Spoke in chat", msg.Text or "", { Focus = false, Reveal = false })
end

function handleAdminCharacterAdded(plr, roleName)
    if addAdminLog then
        addAdminLog(string.format("[%s] %s spawned", roleName, plr.Name), Color3.fromRGB(200, 180, 100), "spawn")
    end
    local adminInfo = onlineAdmins[plr.UserId]
    updateAdminPresence(plr, roleName, adminInfo and adminInfo.rank, "Spawned in the server", nil, { Focus = false, Reveal = false })
end

function trackAdminChat(plr, roleName)
    if adminChatConnections[plr.UserId] then return end
    local ok, channel = pcall(function()
        return TextChatService.TextChannels:WaitForChild("RBXGeneral", 5)
    end)
    if not ok or not channel then return end
    adminChatConnections[plr.UserId] = channel.MessageReceived:Connect(function(msg)
        handleAdminChatMessage(plr, roleName, msg)
    end)
    plr.CharacterAdded:Connect(function()
        handleAdminCharacterAdded(plr, roleName)
    end)
end

function handleStaffPlayerRemoved(plr, roleName)
    if not plr or not roleName then return end
    onlineAdmins[plr.UserId] = nil
    if updateAdminCount then updateAdminCount() end
    if addAdminLog then
        addAdminLog(string.format("LEFT: %s | %s", plr.Name, roleName), Color3.fromRGB(150, 100, 50), "left")
    end
    if adminChatConnections[plr.UserId] then
        adminChatConnections[plr.UserId]:Disconnect()
        adminChatConnections[plr.UserId] = nil
    end
    removeAdminPresence(plr)
end

function handleStaffPlayerJoined(plr, roleName, rank)
    if not plr or not roleName then return end
    onlineAdmins[plr.UserId] = { name = plr.Name, role = roleName, rank = rank }
    if updateAdminCount then updateAdminCount() end
    MIDNIGHT:Notify({
        Title = "Staff Alert",
        Content = string.format("[%d] %s - %s", rank, plr.Name, roleName),
        Type = "warning",
        Duration = 6,
    })
    if addAdminLog then
        addAdminLog(string.format("JOINED: %s | Role: %s | Rank: %d", plr.Name, roleName, rank), Color3.fromRGB(255, 100, 50), "join", true)
    end
    updateAdminPresence(plr, roleName, rank, "Joined the server", nil, { Focus = true, Reveal = true })
    trackAdminChat(plr, roleName)
    plr.AncestryChanged:Connect(function()
        if not plr:IsDescendantOf(game) then
            handleStaffPlayerRemoved(plr, roleName)
        end
    end)
end

function processStaffPlayerCheck(plr)
    local ok, rank = pcall(plr.GetRankInGroup, plr, GROUP_ID)
    if not ok or not rank or rank <= 0 then return end
    local roleName = STAFF_RANKS[rank]
    if roleName then
        handleStaffPlayerJoined(plr, roleName, rank)
    end
end

function checkPlayerForStaff(plr)
    if not staffAlertEnabled then return end
    if plr == player then return end
    if checkedPlayers[plr.UserId] then return end
    checkedPlayers[plr.UserId] = true
    processStaffPlayerCheck(plr)
end

function iterateCurrentPlayersForStaff()
    local currentPlayers = Players:GetPlayers()
    for index = 1, #currentPlayers do
        checkPlayerForStaff(currentPlayers[index])
    end
end

function checkAllCurrentPlayers()
    iterateCurrentPlayersForStaff()
end

formatRemaining = function(seconds)
    seconds = math.max(0, math.floor(tonumber(seconds) or 0))
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = seconds % 60
    if hours > 0 then
        return string.format("%dh %02dm", hours, minutes)
    elseif minutes > 0 then
        return string.format("%dm %02ds", minutes, secs)
    end
    return string.format("%ds", secs)
end

parsePercentValue = function(text)
    local raw = tostring(text or ""):match("(%d+%.?%d*)")
    local value = tonumber(raw)
    if value then
        return math.clamp(value, 0, 100)
    end
    return nil
end

getWorldPoint = function(inst)
    if not inst then
        return nil
    end
    if inst:IsA("BasePart") then
        return inst.Position
    end
    local cached = siteMonitorWallAnchors[inst]
    if cached and cached.Parent then
        return cached.Position
    end
    if inst:IsA("Model") and inst.PrimaryPart then
        siteMonitorWallAnchors[inst] = inst.PrimaryPart
        return inst.PrimaryPart.Position
    end
    local anchor = inst:FindFirstChildWhichIsA("BasePart", true)
    if anchor then
        siteMonitorWallAnchors[inst] = anchor
        return anchor.Position
    end
    return nil
end

findNearestWallScreen = function()
    local wallScreens = workspace:FindFirstChild("WallScreens")
    if not wallScreens then
        return nil
    end

    local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    local origin = root and root.Position or camera.CFrame.Position
    local nearest, nearestDist = nil, math.huge

    for _, wallScreen in ipairs(wallScreens:GetChildren()) do
        local pos = getWorldPoint(wallScreen)
        if pos then
            local dist = (pos - origin).Magnitude
            if dist < nearestDist then
                nearest = wallScreen
                nearestDist = dist
            end
        end
    end

    return nearest
end

    local now = tick()

        refs.AlarmStatus = getChildChain(refs.WallScreen, { "Screen", "Left", "StatusBoardLeft", "statusboard", "status", "status" })

    end

    local notificationRoot = getChildChain(workspace, { "site_monitor", "Part", "StatusBoard", "statusboard", "notification" })
    local eventPart = getChildChain(workspace, { "site_monitor", "Part" })
    if eventPart and eventPart:IsA("BasePart") then
        refs.EventPos = eventPart.Position
    end
    local notification = notificationRoot
    refs.Event076 = notification and getChildChain(notification, { "076", "status" }) or nil
    refs.Event610 = notification and getChildChain(notification, { "610", "status" }) or nil
    refs.EventCDC = notification and getChildChain(notification, { "cdc", "status" }) or nil
end

requestMonitorStreaming = function()
    local now = tick()
    if now < siteMonitorStreamNext then
        return
    end
    siteMonitorStreamNext = now + 8

    if typeof(player.RequestStreamAroundAsync) ~= "function" then
        return
    end

    local refs = siteMonitorRefs
    local targets = { refs.WallScreenPos, refs.EventPos }
    for _, pos in ipairs(targets) do
        if typeof(pos) == "Vector3" then
            task.spawn(function()
                pcall(function()
                    player:RequestStreamAroundAsync(pos)
                end)
            end)
        end
    end
end

updateSimpleMonitorCache = function(label, cacheEntry)
    local text, color = readMonitorLabel(label)
    if text and text ~= "" then
        cacheEntry.Text = text
        cacheEntry.Color = color or cacheEntry.Color
        return true
    end
    return false
end

updateEventMonitorCache = function(label, cacheEntry)
    local text, color = readMonitorLabel(label)
    if text and text ~= "" then
        cacheEntry.Text = text
        cacheEntry.Color = color or cacheEntry.Color
        local pct = parsePercentValue(text)
        if pct then
            cacheEntry.Percent = pct
            if pct >= 100 then
                cacheEntry.ReadyAt = nil
            else
                cacheEntry.ReadyAt = tick() + cacheEntry.Duration * ((100 - pct) / 100)
            end
        end
        return true
    end
    return false
end

getEventCooldownText = function(cacheEntry)
    local readyAt = cacheEntry.ReadyAt
    if readyAt then
        local remaining = math.max(0, readyAt - tick())
        if remaining <= 0.5 then
            cacheEntry.ReadyAt = nil
            return "ready"
        end
        return formatRemaining(remaining)
    end
    if cacheEntry.Percent and cacheEntry.Percent >= 100 then
        return "ready"
    end
    return "--"
end

updateSiteMonitorStage = function()
    refreshSiteMonitorRefs(false)
    requestMonitorStreaming()

    local refs = siteMonitorRefs
    siteMonitorStage = siteMonitorStage % 5 + 1

    if siteMonitorStage == 1 then
        updateSimpleMonitorCache(refs.AlarmStatus, siteMonitorCache.Status)
    elseif siteMonitorStage == 2 then
        updateSimpleMonitorCache(refs.ThreatLevel, siteMonitorCache.Threat)
    elseif siteMonitorStage == 3 then
        updateEventMonitorCache(refs.Event076, siteMonitorCache.Events["076"])
    elseif siteMonitorStage == 4 then
        updateEventMonitorCache(refs.Event610, siteMonitorCache.Events["610"])
    else
        updateEventMonitorCache(refs.EventCDC, siteMonitorCache.Events["CDC"])
    end
end

addChatMessage = function(sender, message, isStaff, isHidden, isLocal)
    local dedupKey = sender .. "|" .. message
    if chatSpyShownDedup[dedupKey] then return end
    chatSpyShownDedup[dedupKey] = true
    task.delay(3, function() chatSpyShownDedup[dedupKey] = nil end)
    chatMsgCount = chatMsgCount + 1
    local nc
    if isStaff then nc = Color3.fromRGB(255, 100, 50)
    elseif isHidden then nc = Color3.fromRGB(100, 200, 100)
    else nc = Color3.fromRGB(150, 180, 255) end
    local prefix = isStaff and "👑" or (isHidden and "🔒" or "💬")
    local label  = prefix .. " " .. sender .. (isHidden and " [HIDDEN]" or "")
    pcall(function()
        if chatLogWd then chatLogWd:AddRichLine(label, message, nc, Color3.fromRGB(160, 160, 160)) end
    end)
    if isStaff and isHidden and addAdminLog then
        addAdminLog(string.format("[HIDDEN CHAT] %s: %s", sender, message), Color3.fromRGB(255, 120, 50), "🔔", true)
    end
    if isHidden and chatSpyEnabled then
        local staffRole = onlineAdmins[Players:FindFirstChild(sender) and Players:FindFirstChild(sender).UserId or 0]
        local spyLabel  = isStaff
            and string.format("[SPY🔴STAFF] %s", sender)
            or  string.format("[SPY] %s", sender)
        pcall(function()
            StarterGui:SetCore("ChatMakeSystemMessage", {
                Color    = isStaff and Color3.fromRGB(255, 80, 30) or Color3.fromRGB(255, 160, 30),
                Font     = Enum.Font.GothamBold,
                TextSize = 16,
                Text     = spyLabel .. ": " .. message,
            })
        end)
    end
end

initChatSpy = function()
    for _, c in ipairs(chatSpyConnections) do pcall(function() c:Disconnect() end) end
    chatSpyConnections = {}; chatSpyShownDedup = {}; chatMsgCount = 0

    -- visibleMessages с TTL: ключи автоматически удаляются через 5 сек,
    -- так словарь не растёт бесконечно за длинную сессию
    local visibleMessages = {}
    local function markVisible(key)
        if visibleMessages[key] then return end  -- уже есть — не перезапускаем таймер
        visibleMessages[key] = true
        task.delay(5, function() visibleMessages[key] = nil end)
    end

    pcall(function()
        local function hookChannel(ch)
            if not ch:IsA("TextChannel") then return end
            local conn = ch.MessageReceived:Connect(function(msg)
                if msg.TextSource then
                    markVisible(tostring(msg.TextSource.UserId) .. "\x00" .. msg.Text)
                end
            end)
            table.insert(chatSpyConnections, conn)
        end
        for _, obj in ipairs(TextChatService:GetChildren()) do hookChannel(obj) end
        local childConn = TextChatService.ChildAdded:Connect(function(obj) hookChannel(obj) end)
        table.insert(chatSpyConnections, childConn)
    end)
    pcall(function()
        local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
        if chatEvents then
            local onMsg = chatEvents:FindFirstChild("OnMessageDoneFiltering")
            if onMsg and onMsg:IsA("RemoteEvent") then
                local conn = onMsg.OnClientEvent:Connect(function(data)
                    if data and type(data.FromSpeaker) == "string" and type(data.Message) == "string" then
                        local plrObj = Players:FindFirstChild(data.FromSpeaker)
                        if plrObj then markVisible(tostring(plrObj.UserId) .. "\x00" .. data.Message) end
                    end
                end)
                table.insert(chatSpyConnections, conn)
            end
        end
    end)
    local function onChatted(targetPlayer, rawMsg)
        if not chatSpyEnabled then return end
        local isLocal = targetPlayer == player
        local isStaff = not isLocal and onlineAdmins[targetPlayer.UserId] ~= nil
        local msg = rawMsg:gsub("[\n\r]", ""):gsub("\t", " "):gsub("[ ]+", " ")
        if #msg > 200 then msg = msg:sub(1, 200) .. "..." end
        task.wait(0.8)
        local key    = tostring(targetPlayer.UserId) .. "\x00" .. msg
        local hidden = not visibleMessages[key] and not isLocal
        addChatMessage(targetPlayer.Name, msg, isStaff, hidden, isLocal)
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        local c = plr.Chatted:Connect(function(msg) task.spawn(onChatted, plr, msg) end)
        table.insert(chatSpyConnections, c)
    end
    local addedConn = Players.PlayerAdded:Connect(function(plr)
        local c = plr.Chatted:Connect(function(msg) task.spawn(onChatted, plr, msg) end)
        table.insert(chatSpyConnections, c)
    end)
    table.insert(chatSpyConnections, addedConn)
    addChatMessage("System", "Chat Spy initialized - monitoring...", false, false, false)
    pcall(function()
        StarterGui:SetCore("ChatMakeSystemMessage", {
            Color = Color3.fromRGB(255, 160, 30), Font = Enum.Font.GothamBold, TextSize = 16,
            Text  = "[SPY] - Active.",
        })
    end)
end

setupPlayerESPListeners = function(plr)
    if plr == player then return end
    local rebuildToken = 0
    local charWatchConn = nil

    local function queueRebuild(delayTime)
        rebuildToken = rebuildToken + 1
        local token = rebuildToken
        task.spawn(function()
            if delayTime and delayTime > 0 then
                task.wait(delayTime)
            end
            local timeoutAt = tick() + 15
            while tick() < timeoutAt do
                if token ~= rebuildToken then
                    return
                end
                local rebuilt = rebuildESP(plr)
                if rebuilt == true then
                    return
                end
                if rebuilt == false then
                    return
                end
                task.wait(0.15)
            end
        end)
    end

    local function hookCharacter(char)
        if charWatchConn then
            pcall(function() charWatchConn:Disconnect() end)
            charWatchConn = nil
        end
        if not char then return end
        charWatchConn = char.DescendantAdded:Connect(function(desc)
            if desc and (desc.Name == "HumanoidRootPart" or desc:IsA("Humanoid")) then
                queueRebuild(0)
            end
        end)
    end

    addPlayerConn(plr, plr.CharacterAdded:Connect(function(char)
        hookCharacter(char)
        queueRebuild(0.2)
    end))
    if plr.Character then
        hookCharacter(plr.Character)
    end
    pcall(function()
        addPlayerConn(plr, plr.CharacterAppearanceLoaded:Connect(function()
            queueRebuild(0)
        end))
    end)
    pcall(function()
        addPlayerConn(plr, plr:GetPropertyChangedSignal("Team"):Connect(function()
            queueRebuild(0.15)
        end))
    end)
end

Players.PlayerAdded:Connect(function(plr)
    setupPlayerESPListeners(plr)
end)
